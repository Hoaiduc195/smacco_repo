\section{Idea}

\subsection{Đặt vấn đề}
Hiện nay, việc tìm kiếm một nơi lưu trú phù hợp khi đi du lịch không hề đơn giản. Người dùng thường phải cân nhắc nhiều yếu tố như giá cả, vị trí, tiện nghi, chất lượng dịch vụ và các đánh giá từ những khách hàng trước đó. Quá trình này vẫn còn gặp nhiều khó khăn với các \textbf{điểm đau (pain points)} cốt lõi:

\begin{itemize}[label=\textcolor{blue!70!black}{\faTimesCircle}]
    \item \textbf{Thông tin phân mảnh \& Rời rạc:} Dữ liệu về khách sạn, homestay rải rác trên nhiều nền tảng, khiến người dùng mất nhiều thời gian tổng hợp và so sánh.
    \item \textbf{Bộ lọc cứng nhắc \& Thiếu cá nhân hoá:} Các công cụ truyền thống chỉ cho phép lọc theo các tùy chọn có sẵn, không hỗ trợ tìm kiếm linh hoạt dựa trên nhu cầu tự nhiên và đặc thù của người dùng.
    \item \textbf{Độ tin cậy của dữ liệu thấp:} Đánh giá từ người dùng trước thường mang tính chủ quan, có thể bị thao túng (fake reviews) và thiếu tính cập nhật thực tế tại thời điểm hiện tại.
    \item \textbf{Quá tải thông tin \& Tốn thời gian:} Người dùng mất quá nhiều thời gian để đọc và đối chiếu thông tin từ nhiều nguồn trước khi có thể đưa ra quyết định chính xác.
\end{itemize}

\begin{infobox}[\faLightbulb \ Giải pháp đề xuất]
Nhóm xây dựng hệ thống tìm kiếm và gợi ý lưu trú thông minh, cho phép người dùng mô tả nhu cầu bằng \textbf{ngôn ngữ tự nhiên}. Hệ thống sẽ phân tích yêu cầu, tìm kiếm các địa điểm phù hợp, đánh giá, xếp hạng và đưa ra những gợi ý trực quan, giúp người dùng ra quyết định nhanh chóng.
\end{infobox}

\subsection{Tầm nhìn ứng dụng}
Dự án không chỉ nhằm xây dựng một nền tảng thương mại thông thường, mà định hình lại cách người dùng tương tác với dữ liệu. Thông qua việc ứng dụng AI và kỹ thuật \textbf{RAG}, hệ thống kỳ vọng tạo ra một tiền đề cho tương lai, nơi AI đóng vai trò như một người điều phối đáng tin cậy.

\subsection{Người dùng mục tiêu}
\begin{itemize}[label=\textcolor{cyan!70!black}{\faUserCheck}]
    \item \textbf{Khách du lịch tự túc:} Cần tìm kiếm nhanh chóng, so sánh giá để tối ưu hóa trải nghiệm.
    \item \textbf{Người dùng On-site:} Đang lưu trú tại địa điểm, sẵn sàng đóng góp thông tin thực tế và trả lời câu hỏi cộng đồng.
    \item \textbf{Người ưu tiên cá nhân hóa:} Thích tương tác với AI thay vì tốn hàng giờ với các bộ lọc phức tạp.
\end{itemize}


\section{Problem Analysis}

\subsection{Hành trình Người dùng}
Thay vì tiếp cận theo hướng kỹ thuật khô khan, nhóm thiết kế hệ thống thông qua chuỗi câu hỏi dẫn dắt từ góc nhìn người dùng:

\begin{tcolorbox}[colback=white, colframe=gray!30, arc=5pt, boxrule=1pt, breakable]
\begin{description}[style=multiline, leftmargin=1cm, font=\bfseries\color{blue!70!black}]
    \item[\faSearch] \textbf{Tìm kiếm thế nào?} Cần giao diện linh hoạt (bản đồ, lọc tiêu chí, từ khóa) kết hợp LLM để phân tích ý định thô.
    \item[\faBalanceScale] \textbf{So sánh ra sao?} Cung cấp bảng đối chiếu tiện ích và nhận xét tổng hợp từ AI để ra quyết định khách quan.
    \item[\faSearchPlus] \textbf{Hiểu sâu địa điểm?} AI khai thác hàng ngàn bài đánh giá, trả lời trực tiếp các câu hỏi chi tiết về wifi, bữa sáng,...
    \item[\faCheckCircle] \textbf{Chống "ảo giác" AI?} Sử dụng dữ liệu từ người dùng "On-site" để đảm bảo tính xác thực tối đa.
\end{description}
\end{tcolorbox}

\subsection{Phân rã Bài toán Tính toán}
Từ nhu cầu thực tế, hệ thống được phân rã thành các module kỹ thuật chuyên biệt:
\begin{itemize}[label=\textcolor{blue!60}{\faCogs}]
    \item \textbf{Intent Parsing:} LLM chuyển đổi ngôn ngữ tự nhiên thành JSON cấu trúc.
    \item \textbf{Data Aggregation:} Truy xuất đồng thời DB nội bộ và API đối tác.
    \item \textbf{Semantic Search:} Số hóa văn bản thành vector embeddings (pgvector).
    \item \textbf{Workflow Engine:} Xây dựng khung làm việc cứng (guardrails) kiểm soát AI.
\end{itemize}

\subsection{Ràng buộc, Giả định và Tiêu chí}
Để đảm bảo tính khả thi trong phạm vi đồ án, hệ thống được thiết kế và đánh giá dựa trên các khía cạnh rõ ràng sau:

\vspace{0.3cm}
\noindent\textbf{\textcolor{blue!80!black}{\faLink \ 1. Ràng buộc}}
\begin{itemize}[label=\textcolor{gray!70!black}{\faCaretRight}]
    \item \textbf{Về mặt kiến trúc:} Lựa chọn mô hình \textit{Modular Monolith} để giảm thiểu chi phí quản lý và vận hành trong giai đoạn MVP, đồng thời vẫn giữ được tính phân tách cao giúp dễ dàng mở rộng sang Microservices sau này.
    \item \textbf{Về mặt tài nguyên:} Hệ thống offload (chuyển giao) toàn bộ tải tính toán AI nặng nề sang các API của bên thứ ba (như Groq). Nhờ đó, hệ thống lõi có thể chạy cục bộ mượt mà trên phần cứng cơ bản không yêu cầu GPU.
    \item \textbf{Về phạm vi hỗ trợ:} Ứng dụng ở giai đoạn này tập trung vào thị trường Việt Nam. Hệ thống tập trung xác định các địa điểm lưu trú trong nước, yêu cầu AI phải hỗ trợ tiếng Việt tốt và hiểu rõ ngữ cảnh văn hóa địa phương.
\end{itemize}

\vspace{0.3cm}
\noindent\textbf{\textcolor{blue!80!black}{\faLightbulb \ 2. Giả định}}
\begin{itemize}[label=\textcolor{gray!70!black}{\faCaretRight}]
    \item Khả năng lập luận, phân tích ngữ nghĩa và đưa ra quyết định của Mô hình ngôn ngữ lớn là yếu tố khách quan, tính chính xác phụ thuộc vào API cung cấp chứ không bị giới hạn bởi hệ thống nội bộ.
    \item Dữ liệu thu thập từ các nhà cung cấp bên thứ ba (Goong, SerpAPI) luôn đảm bảo độ tin cậy về mặt nội dung và ổn định về thời gian phản hồi (uptime).
\end{itemize}

\vspace{0.3cm}
\noindent\textbf{\textcolor{blue!80!black}{\faTasks \ 3. Tiêu chí đánh giá}}
\begin{itemize}[label=\textcolor{gray!70!black}{\faCaretRight}]
    \item \textbf{Sự hài lòng của người dùng:} Đánh giá mức độ thoải mái, mức độ cá nhân hóa và sự hài lòng tổng thể khi người dùng trải nghiệm các tính năng tìm kiếm, trò chuyện cùng AI trên nền tảng.
    \item \textbf{Tính hữu dụng:} Ứng dụng phải giải quyết được triệt để bài toán tìm kiếm chỗ ở. Giao diện cần trực quan, dễ thao tác và có luồng tương tác mượt mà phù hợp với nhiều nhóm đối tượng.
    \item \textbf{Độ chính xác của thông tin:} Đảm bảo chất lượng dữ liệu trả về. Danh sách địa điểm phải sát thực tế, và đặc biệt câu trả lời phân tích chi tiết từ AI phải chuẩn xác, tuyệt đối không được xảy ra tình trạng ảo giác.
\end{itemize}

\section{Pattern Recognition}

Để thiết kế một kiến trúc hệ thống chặt chẽ, nhóm đã tiến hành phân tích và nhận diện các hành vi lặp lại của người dùng cũng như các cấu trúc tiềm ẩn trong luồng dữ liệu thực tế.

\vspace{0.4cm}
\noindent\textcolor{blue!80!black}{\large\textbf{\faSearch \ 1. User Query Pattern}}
\vspace{0.15cm}

\begin{tcolorbox}[blanker, borderline west={3pt}{0pt}{blue!60!black}, left=12pt, top=2pt, bottom=2pt]
Quan sát thực tế cho thấy, dù người dùng có cách diễn đạt rất đa dạng khi tìm kiếm nơi lưu trú, các câu truy vấn này luôn xoay quanh một tập hợp các đặc trưng cốt lõi lặp lại. Những yếu tố này bao gồm các thông số định lượng như vị trí, ngân sách, loại hình lưu trú, tiện nghi, và các yếu tố định tính (yêu cầu mềm) như "yên tĩnh", "gần trung tâm" hay "phù hợp cho gia đình". Nhận diện được mẫu cấu trúc tiềm ẩn này, \textbf{quy tắc tổng quát} được rút ra là: mọi câu lệnh phức tạp đều có thể được bóc tách thành các tham số rời rạc kết hợp với ngữ nghĩa mở rộng. Quy tắc này trực tiếp định hình quyết định thiết kế hệ thống phân tích ý định (intent parsing). Thay vì cố gắng khớp chuỗi nguyên văn, kiến trúc hệ thống sử dụng mô hình ngôn ngữ để trích xuất và định tuyến các thông tin này vào một cấu trúc tham số chuẩn hóa, làm đầu vào cho các bước truy vấn tiếp theo.
\end{tcolorbox}

\vspace{0.4cm}
\noindent\textcolor{teal!80!black}{\large\textbf{\faDatabase \ 2. Accommodation Data Pattern}}
\vspace{0.15cm}

\begin{tcolorbox}[blanker, borderline west={3pt}{0pt}{teal!60!black}, left=12pt, top=2pt, bottom=2pt]
Trong quá trình thu thập thông tin từ nhiều nhà cung cấp khác nhau, nhóm phát hiện ra một sự tương đồng lớn về cấu trúc dữ liệu. Dù định dạng đầu vào có khác biệt, mọi điểm lưu trú đều chia sẻ các thuộc tính chung cốt lõi như tên, tọa độ địa lý, địa chỉ, mức giá, điểm đánh giá, tiện nghi, loại hình và các bài nhận xét. Mẫu cấu trúc này chỉ ra rằng, dữ liệu đa nguồn hoàn toàn có thể được ánh xạ về một khuôn dạng chung. Dựa trên quy tắc này, nhóm đã đưa ra \textbf{quyết định thiết kế} một lược đồ dữ liệu và các đối tượng truyền tải dữ liệu (DTO) thống nhất. Mọi luồng dữ liệu từ Goong, SerpAPI hay cơ sở dữ liệu nội bộ đều phải đi qua bước chuẩn hóa này, đảm bảo hệ thống có thể xử lý và hiển thị thông tin nhất quán bất kể nguồn gốc.
\end{tcolorbox}

\vspace{0.4cm}
\noindent\textcolor{orange!80!black}{\large\textbf{\faBalanceScale \ 3. Decision-making Pattern}}
\vspace{0.15cm}

\begin{tcolorbox}[blanker, borderline west={3pt}{0pt}{orange!70!black}, left=12pt, top=2pt, bottom=2pt]
Hành vi của người dùng trong quá trình lựa chọn chỗ ở hiếm khi là một đường thẳng. Quan sát cho thấy, du khách thường không chọn ngay kết quả đầu tiên được đề xuất. Thay vào đó, họ có xu hướng lưu lại một vài lựa chọn nổi bật, đặt chúng lên bàn cân để so sánh các yếu tố như giá cả, vị trí và tiện ích trước khi đưa ra quyết định cuối cùng. Nhận ra mẫu hành vi vòng lặp này, \textbf{quy tắc tổng quát} được xác định là quá trình tìm kiếm luôn đi kèm với nhu cầu đối chiếu đa chiều. Từ đó, hệ thống được thiết kế bổ sung một luồng tính năng chuyên biệt cho việc đánh dấu và lưu trữ địa điểm. Điều này giúp trải nghiệm không bị gián đoạn và hỗ trợ người dùng ra quyết định một cách tự tin hơn.
\end{tcolorbox}

\vspace{0.4cm}
\noindent\textcolor{purple!80!black}{\large\textbf{\faStar \ 4. Review và RAG Pattern}}
\vspace{0.15cm}

\begin{tcolorbox}[blanker, borderline west={3pt}{0pt}{purple!60!black}, left=12pt, top=2pt, bottom=2pt]
Khi đi sâu vào các câu hỏi mà người dùng thường đặt ra về một khách sạn cụ thể, một mẫu chủ đề rõ nét xuất hiện. Hầu hết các thắc mắc đều lặp lại quanh các khía cạnh: độ sạch sẽ, chất lượng mạng kết nối, mức độ tiếng ồn, vị trí thực tế so với quảng cáo và thái độ phục vụ. Từ sự hội tụ về chủ đề này, \textbf{quy tắc rút ra} là phần lớn giá trị của các bài đánh giá nằm ở việc trả lời các câu hỏi cụ thể này. Nhận thức được điều đó, thay vì để người dùng đọc toàn bộ văn bản dài, nhóm \textbf{quyết định thiết kế} tổ chức dữ liệu đánh giá thành các đoạn thông tin nhỏ (chunks) và sử dụng kỹ thuật tìm kiếm vector. Thiết kế này cho phép hệ thống tự động trích xuất đúng phần nhận xét chứa câu trả lời và tổng hợp lại cho người dùng.
\end{tcolorbox}

\vspace{0.4cm}
\noindent\textcolor{red!80!black}{\large\textbf{\faExclamationTriangle \ 5. Failure Pattern}}
\vspace{0.15cm}

\begin{tcolorbox}[blanker, borderline west={3pt}{0pt}{red!60!black}, left=12pt, top=2pt, bottom=2pt]
Hoạt động của các dịch vụ tích hợp bên ngoài luôn tiềm ẩn những rủi ro có tính quy luật. Quá trình kiểm thử cho thấy các lỗi thường lặp lại theo những dạng nhất định: thiếu thông tin về giá, không có điểm đánh giá, dữ liệu trả về bị trùng lặp, lỗi kết nối hoặc hoàn toàn không tìm thấy kết quả phù hợp. Mẫu lỗi này dẫn đến \textbf{quy tắc}: sự thiếu hụt hoặc sai lệch dữ liệu ngoại vi là điều tất yếu và hệ thống lõi phải có khả năng tự phục hồi. Áp dụng quy tắc này vào \textbf{quyết định thiết kế}, các cơ chế xử lý lỗi biên, khử trùng lặp (deduplication) và kịch bản dự phòng (fallback) đã được xây dựng. Khi một nhà cung cấp gặp sự cố hoặc dữ liệu không toàn vẹn, hệ thống sẽ tự động bù đắp bằng các luồng thông tin khác hoặc cấu trúc lại giao diện phù hợp mà không làm sập toàn bộ ứng dụng.

\subsection{Biểu diễn Bài toán}

\textbf{Khái niệm:} Biểu diễn bài toán là quá trình chuyển đổi bài toán thực tế (xây dựng nền tảng khám phá chỗ ở thông minh) thành mô hình, cấu trúc dữ liệu và ký hiệu mà máy tính có thể xử lý.

\textbf{Cách nhóm áp dụng 5 dạng biểu diễn:}

\begin{enumerate}
    \item \textbf{Biểu diễn dữ liệu:} Các thực thể thực tế (người dùng, địa điểm, đánh giá, hội thoại, tệp tin) được ánh xạ thành 13 mô hình quan hệ trong PostgreSQL. Mỗi thực thể được gán kiểu dữ liệu tương ứng: UUID cho định danh, JSONB cho dữ liệu linh hoạt, vector cho nhúng ngữ nghĩa.
    \item \textbf{Biểu diễn cấu trúc:} Mối quan hệ giữa người dùng---địa điểm---nội dung được mô hình hóa qua ERD với khóa ngoại và chỉ mục. Sự phụ thuộc giữa các module được biểu diễn bằng sơ đồ lớp và C4 Model.
    \item \textbf{Biểu diễn quy trình:} Luồng tương tác (tìm kiếm $\to$ gợi ý $\to$ trò chuyện AI $\to$ check-in) được mô hình hóa qua sơ đồ tuần tự và flowchart.
    \item \textbf{Biểu diễn toán học:} Công thức tính điểm gợi ý đa yếu tố (rating, ngân sách, khoảng cách, tiện ích, cận kề) với trọng số và hàm chuẩn hóa.
    \item \textbf{Biểu diễn logic:} Quy tắc định tuyến ý định người dùng (IF-THEN) qua LLM JSON Mode; logic kiểm tra điều kiện trong workflow registry.
\end{enumerate}

\textbf{Ba nguyên tắc cốt lõi được tuân thủ:}
\begin{itemize}
    \item \textbf{Trừu tượng hóa:} Chỉ lưu nội bộ địa điểm người dùng tương tác, ủy thác khám phá cho API ngoài --- loại bỏ chi tiết không cần thiết.
    \item \textbf{Mô hình hóa có cấu trúc:} Mỗi module có ranh giới rõ ràng (controller/service/DTO), dữ liệu và quan hệ được đồng bộ qua Prisma schema.
    \item \textbf{Đa biểu diễn:} Cùng một bài toán (tìm kiếm chỗ ở) được biểu diễn đồng thời dưới dạng flowchart (luồng), công thức (điểm số) và mã giả (giải thuật hợp nhất).
\end{itemize}

\subsection{Di trú Kiến trúc: Microservices $\to$ Modular Monolith}

Hệ thống nguyên bản vận hành với bốn dịch vụ riêng rẽ (Nginx Gateway, Core-service, AI-service, Recommendation-service). Kiến trúc này bộc lộ các vấn đề:

\begin{itemize}
    \item \textbf{Chi phí mạng:} Mỗi yêu cầu chat trải qua ba chặng HTTP, gây độ trễ tích lũy.
    \item \textbf{Trùng lặp logic:} Xác thực, xử lý lỗi, cấu hình bị sao chép giữa các dịch vụ.
    \item \textbf{Phân mảnh kiểu:} Python $\leftrightarrow$ TypeScript yêu cầu tuần tự hóa thủ công, dễ sai sót.
    \item \textbf{Phức tạp vận hành:} Bốn Docker image, bốn pipeline triển khai riêng biệt.
\end{itemize}

\textbf{Giải pháp:} Hợp nhất thành một monolith dạng module NestJS duy nhất. Các lời gọi HTTP liên dịch vụ được thay thế bằng phương thức gọi trực tiếp qua dependency injection, loại bỏ Gateway Nginx trung gian.

\subsection{Các Thách thức và Giải pháp}

\subsubsection{Tính phi tất định của LLM}

\textbf{Thách thức:} LLM sinh phản hồi không nhất quán về định dạng, dễ hallucination dữ liệu, và độ trễ khó dự đoán.

\textbf{Giải pháp --- Kiến trúc Bốn Lớp:}
\begin{itemize}
    \item \textbf{Lớp 1 (Router):} LLM với JSON Mode $\to$ phân loại ý định, trích xuất tham số có cấu trúc. Thất bại $\to$ mặc định GENERAL\_CHAT.
    \item \textbf{Lớp 2 (Workflow Engine):} Thực thi DAG các bước tất định, không gọi LLM.
    \item \textbf{Lớp 3 (Tool Layer):} Mỗi công cụ có hợp đồng đầu ra chuẩn hóa, không ném ngoại lệ.
    \item \textbf{Lớp 4 (Composer):} Neo phản hồi Markdown vào dữ liệu thực tế từ công cụ, giảm hallucination.
\end{itemize}

Lợi ích: Workflow mới có thể thêm dưới dạng JSON, không cần biên dịch lại.

\subsubsection{Streaming SSE qua POST}

\textbf{Thách thức:} SSE chuẩn chỉ hỗ trợ GET, trong khi endpoint chat cần gửi JSON payload và token xác thực.

\textbf{Giải pháp --- Mô hình hai pha:}
\begin{itemize}
    \item Sử dụng \texttt{fetch()} với \texttt{ReadableStream} thay vì \texttt{EventSource}, cho phép POST và đính kèm token.
    \item Pha 1: Gửi dữ liệu cấu trúc (\texttt{searchAction}) ngay sau khi công cụ thực thi --- cho phép frontend render bản đồ trước.
    \item Pha 2: Truyền token LLM dưới dạng delta SSE.
    \item Header \texttt{X-Accel-Buffering: no} để vô hiệu hóa bộ đệm Nginx khi triển khai sau reverse proxy.
\end{itemize}

\subsubsection{Tích hợp Đa Nhà cung cấp Bản đồ}

\textbf{Thách thức:} Mỗi API bản đồ trả về định dạng khác nhau; lỗi một nhà cung cấp không được ảnh hưởng toàn bộ tìm kiếm.

\textbf{Giải pháp:}
\begin{itemize}
    \item Chuẩn hóa giao diện đầu ra cho mọi nhà cung cấp.
    \item Gọi song song qua \texttt{Promise.all} với \texttt{.catch()} riêng biệt --- đảm bảo cách ly lỗi.
    \item Chuỗi dự phòng: Goong $\to$ OSM $\to$ mảng rỗng.
\end{itemize}

\subsubsection{Vector Embedding và Prisma}

\textbf{Thách thức:} Prisma chưa hỗ trợ nguyên sinh kiểu \texttt{vector} của PostgreSQL.

\textbf{Giải pháp:} Khai báo cột \texttt{embedding} dưới dạng \texttt{Unsupported("vector")}, thao tác qua SQL thô (\texttt{\$queryRawUnsafe}).

\subsubsection{Xác thực Firebase}

\textbf{Thách thức:} Tích hợp Firebase JWT vào vòng đời NestJS guard mà không sao chép logic xác thực trên mọi controller.

\textbf{Giải pháp:} Triển khai guard với \texttt{CanActivate}: trích xuất Authorization header $\to$ xác thực token qua Firebase Admin SDK $\to$ gắn decoded token vào request. Frontend Axios interceptor tự động đính kèm token.

\subsection{Các Thách thức Khác}

\textbf{Xử lý đồng thời:} Ba mẫu thiết kế được áp dụng: (1) \texttt{Promise.all} với catch riêng biệt; (2) ghi vào khóa khác nhau trong shared state để tránh race condition; (3) Active Flag pattern để tránh cập nhật state sau khi component unmount.

\textbf{Triển khai Docker:} Ba container (frontend/Nginx, backend/Node 22, PostgreSQL/pgvector) với chiến lược build đa tầng tận dụng layer cache. Docker override cho môi trường phát triển sử dụng volume ẩn danh tránh xung đột \texttt{node\_modules}.

\textbf{Quản lý bản đồ frontend:} Chuyển đổi layer bằng opacity, MarkerClusterGroup với chunkedLoading cho hiệu năng, CustomEvent thay vì prop drilling cho tích hợp AI-Search.

\subsection{Kết luận}

Quá trình triển khai cho thấy việc chuyển đổi từ mô hình thiết kế sang mã nguồn đòi hỏi liên tục giải quyết các vấn đề thực tế: tính phi tất định của LLM, hạn chế của ORM với kiểu dữ liệu đặc thù, và đảm bảo tính nhất quán xuyên suốt các tầng. Mười hai vấn đề đã được ghi nhận với các mức độ ưu tiên khác nhau.

\subsection{Những kiến thức và kỹ năng đạt được}
Môn học và đồ án lần này đã mang lại cho chúng em những bài học vô cùng thực tế, giúp thu hẹp khoảng cách giữa lý thuyết và ứng dụng:
\begin{itemize}
    \item \textbf{Vận dụng 4 trụ cột cốt lõi của Tư duy Máy tính (Computational Thinking):} Thông qua đồ án, chúng em đã làm chủ và áp dụng thành công mô hình tư duy này để giải quyết bài toán thực tế. Cụ thể, chúng em thực hiện 
    \begin{enumerate}
        \item \textit{Phân rã bài toán} (Decomposition) để chia nhỏ hệ thống thành các module độc lập (\textit{AiModule} và \textit{RecommendationsModule});
        \item \textit{Nhận biết mẫu} (Pattern Recognition) để tìm ra khuôn mẫu trong hành vi tìm kiếm của khách hàng;
        \item \textit{Trừu tượng hóa} (Abstraction) để lọc bỏ thông tin nhiễu, trích xuất tham số cốt lõi qua \textit{Intent Parsing};
        \item \textit{Thiết kế thuật toán} (Algorithm Design) để xây dựng luồng vận hành (Workflow) tối ưu cho Chatbot.
    \end{enumerate}
       \noindent Các kỹ năng này giúp chúng em định hình được tư duy logic vững chắc cho các dự án phức tạp sau này.
    
    \item \textbf{Tư duy khai thác Trí tuệ nhân tạo: }Chúng em đã học được cách viết cấu trúc câu lệnh (Prompt Engineering) và cách ứng dụng AI một cách hiệu quả, biến AI thành một công cụ đồng hành đắc lực để tối ưu hóa việc tìm kiếm thông tin thay vì lạm dụng một cách máy móc.
    \item \textbf{Làm quen với các công nghệ mới: }Đồ án là cơ hội để chúng em tiếp cận và biết cách sử dụng hàng loạt công cụ, ngôn ngữ và nền tảng cốt lõi trong phát triển phần mềm hiện đại như HTML, JavaScript, nền tảng cơ sở dữ liệu Firebase, bản đồ OpenStreetMap,... Dù thời gian có hạn và chưa thể tìm hiểu quá sâu sắc từng công nghệ, nhưng đây là nền móng cực kỳ quan trọng cho các môn học chuyên ngành tiếp theo.
    \item \textbf{Kỹ năng làm việc nhóm và Quản lý mã nguồn: }Biết cách sử dụng GitHub để quản lý các phiên bản code, phối hợp phân chia công việc trong nhóm một cách hệ thống.
    \item \textbf{Tư duy Hệ thống: }Hiểu rõ bức tranh toàn cảnh về cách một hệ thống phần mềm vận hành trong thực tế (cách Frontend tương tác với Backend, cách gọi API và truy vấn dữ liệu từ Database).
\end{itemize}

\subsection{Hướng phát triển trong tương lai}
Nhằm nâng cấp và hoàn thiện dự án thành một sản phẩm có tính ứng dụng thực tế cao hơn, các mục tiêu trong tương lai bao gồm:
\begin{itemize}
    \item \textbf{Nghiên cứu sâu về RAG và LLM: }Tối ưu hóa sâu hơn kỹ thuật Retrieval-Augmented Generation (RAG) kết hợp với các mô hình ngôn ngữ lớn (như Groq) để nâng cao độ chính xác, giảm thiểu hiện tượng "ảo tưởng" (hallucination) của Chatbot khi trả lời khách hàng.
    \item \textbf{Tối ưu hóa UI/UX: }Thiết kế lại giao diện người dùng thân thiện, trực quan hơn trên cả hai nền tảng Web và Mobile, tối ưu hóa các bộ lọc nâng cao để nâng cao trải nghiệm của các nhóm đối tượng Target Users.
\end{itemize}
