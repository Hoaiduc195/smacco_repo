\section{Idea}

\subsection{Problem Statement}
Hiện nay, việc tìm kiếm một nơi lưu trú phù hợp khi đi du lịch không hề đơn giản. Người dùng thường phải cân nhắc nhiều yếu tố như giá cả, vị trí, tiện nghi, chất lượng dịch vụ và các đánh giá từ những khách hàng trước đó. Quá trình này vẫn còn gặp nhiều khó khăn với các \textbf{điểm đau} cốt lõi:

\begin{itemize}[label=\textcolor{blue!70!black}{\faTimesCircle}]
  \item \textbf{Thông tin phân mảnh và Rời rạc:} Dữ liệu về khách sạn, homestay rải rác trên nhiều nền tảng, khiến người dùng mất nhiều thời gian tổng hợp và so sánh.
  \item \textbf{Bộ lọc cứng nhắc và Thiếu cá nhân hoá:} Các công cụ truyền thống chỉ cho phép lọc theo các tùy chọn có sẵn, không hỗ trợ tìm kiếm linh hoạt dựa trên nhu cầu tự nhiên và đặc thù của người dùng.
  \item \textbf{Độ tin cậy của dữ liệu thấp:} Đánh giá từ người dùng trước thường mang tính chủ quan, có thể bị thao túng bằng các đánh giá giả mạo và thiếu tính cập nhật thực tế tại thời điểm hiện tại.
  \item \textbf{Quá tải thông tin và Tốn thời gian:} Người dùng mất quá nhiều thời gian để đọc và đối chiếu thông tin từ nhiều nguồn trước khi có thể đưa ra quyết định chính xác.
\end{itemize}

\begin{infobox}[\faLightbulb \ Giải pháp đề xuất]
Nhóm xây dựng hệ thống tìm kiếm và gợi ý lưu trú thông minh, cho phép người dùng mô tả nhu cầu bằng \textbf{ngôn ngữ tự nhiên}. Hệ thống sẽ phân tích yêu cầu, tìm kiếm các địa điểm phù hợp, đánh giá, xếp hạng và đưa ra những gợi ý trực quan, giúp người dùng ra quyết định nhanh chóng.
\end{infobox}

\subsection{Application Vision}
Dự án không chỉ nhằm xây dựng một nền tảng thương mại thông thường, mà định hình lại cách người dùng tương tác với dữ liệu. Thông qua việc ứng dụng AI và kỹ thuật \textbf{RAG}, hệ thống kỳ vọng tạo ra một tiền đề cho tương lai, nơi AI đóng vai trò như một người điều phối đáng tin cậy.

\subsection{Target Users}
\begin{itemize}[label=\textcolor{cyan!70!black}{\faUserCheck}]
  \item \textbf{Khách du lịch tự túc:} Cần tìm kiếm nhanh chóng, so sánh giá để tối ưu hóa trải nghiệm.
  \item \textbf{Người dùng On-site:} Đang lưu trú tại địa điểm, sẵn sàng đóng góp thông tin thực tế và trả lời câu hỏi cộng đồng.
  \item \textbf{Người ưu tiên cá nhân hóa:} Thích tương tác với AI thay vì tốn hàng giờ với các bộ lọc phức tạp.
\end{itemize}


\section{Problem Analysis}

\subsection{User Journey}
Thay vì tiếp cận theo hướng kỹ thuật khô khan, nhóm thiết kế hệ thống thông qua chuỗi câu hỏi dẫn dắt từ góc nhìn người dùng:

\begin{itemize}[label=\textcolor{blue!70!black}{\faQuestionCircle}, leftmargin=0.5cm]
  \item \textbf{Người dùng sẽ làm gì với ứng dụng của tôi?} \\
  $\rightarrow$ Người dùng cần tìm địa điểm lưu trú.
  
  \item \textbf{Người dùng có thể tìm kiếm địa điểm lưu trú như thế nào?} \\
  $\rightarrow$ Thiết kế giao diện tìm kiếm hoặc phương thức tìm kiếm linh hoạt.

  \item \textbf{Vậy làm sao để phương thức tìm kiếm có thể cho ra kết quả tốt?} \\
  $\rightarrow$ Chúng ta sẽ đánh giá trên các tiêu chí, như vậy chúng ta cần trích xuất ý định của người dùng.

  \item \textbf{Vậy việc trích xuất nên được thực hiện thế nào?} \\
  $\rightarrow$ Có thể cho phép người dùng chọn hoặc thông qua tác tử AI được cài đặt.

  \item \textbf{Nếu người dùng sau khi đã có các địa điểm phù hợp với nhu cầu của họ, làm cách nào để hệ thống có thể giúp người dùng ra quyết định cuối cùng như thế nào?} \\
  $\rightarrow$ Có thể cài đặt tính năng giúp người dùng so sánh các địa điểm, hỗ trợ ra quyết định cuối.

  \item \textbf{Vậy nếu người dùng muốn tìm hiểu sâu hơn về một địa điểm thì sao?} \\
  $\rightarrow$ Cài đặt tính năng AI đưa ra phân tích chi tiết về một địa điểm cụ thể, cho phép người dùng lựa chọn các tiêu chí để phân tích.

  \item \textbf{Việc sử dụng LLM có thể không đạt được độ chính xác cao, liệu nguồn thông tin nào uy tín hơn không?} \\
  $\rightarrow$ Sử dụng nguồn thông tin từ chính người dùng, có thể áp dụng chính sách 'người dùng onsite' để tăng độ tin cậy.

  \item \textbf{Vậy với các tính năng như trên, có thể tối giản giao diện để người dùng có thể tương tác nhanh chóng và dễ dàng, phù hợp với nhiều người dùng ở các độ tuổi, trình độ khác nhau hay không?} \\
  $\rightarrow$ Sử dụng LLM đóng vai trò điều phối trung gian, người dùng sẽ tương tác trực tiếp với LLM, LLM sẽ trích xuất ý định người dùng và tiến hành các yêu cầu một cách tự động.
\end{itemize}

\subsection{Computational Problem Decomposition}
Từ nhu cầu thực tế, hệ thống được phân rã thành các module kỹ thuật chuyên biệt:
\begin{itemize}[label=\textcolor{blue!60}{\faCogs}]
  \item \textbf{Intent Parsing:} LLM chuyển đổi ngôn ngữ tự nhiên thành JSON cấu trúc.
  \item \textbf{Data Aggregation:} Truy xuất đồng thời DB nội bộ và API đối tác.
  \item \textbf{Semantic Search:} Số hóa văn bản thành vector nhúng (với pgvector).
  \item \textbf{Workflow Engine:} Xây dựng khung làm việc cứng kiểm soát AI.
\end{itemize}

\subsection{Constraints, Assumptions, and Criteria}
Để đảm bảo tính khả thi trong phạm vi đồ án, hệ thống được thiết kế và đánh giá dựa trên các khía cạnh rõ ràng sau:

\vspace{0.3cm}
\noindent\textbf{\textcolor{blue!80!black}{\faLink \ 1. Ràng buộc}}
\begin{itemize}[label=\textcolor{gray!70!black}{\faCaretRight}]
  \item \textbf{Về mặt kiến trúc:} Lựa chọn mô hình \textit{Modular Monolith} để giảm thiểu chi phí quản lý và vận hành trong giai đoạn MVP, đồng thời vẫn giữ được tính phân tách cao giúp dễ dàng mở rộng sang Microservices sau này.
  \item \textbf{Về mặt tài nguyên:} Hệ thống chuyển giao toàn bộ tải tính toán AI nặng nề sang các API của bên thứ ba (như Groq). Nhờ đó, hệ thống lõi có thể chạy cục bộ mượt mà trên phần cứng cơ bản không yêu cầu GPU.
  \item \textbf{Về phạm vi hỗ trợ:} Ứng dụng ở giai đoạn này tập trung vào thị trường Việt Nam. Hệ thống tập trung xác định các địa điểm lưu trú trong nước, yêu cầu AI phải hỗ trợ tiếng Việt tốt và hiểu rõ ngữ cảnh văn hóa địa phương.
\end{itemize}

\vspace{0.3cm}
\noindent\textbf{\textcolor{blue!80!black}{\faLightbulb \ 2. Giả định}}
\begin{itemize}[label=\textcolor{gray!70!black}{\faCaretRight}]
  \item Khả năng lập luận, phân tích ngữ nghĩa và đưa ra quyết định của Mô hình ngôn ngữ lớn là yếu tố khách quan, tính chính xác phụ thuộc vào API cung cấp chứ không bị giới hạn bởi hệ thống nội bộ.
  \item Dữ liệu thu thập từ các nhà cung cấp bên thứ ba (Goong, SerpAPI) luôn đảm bảo độ tin cậy về mặt nội dung và ổn định về thời gian hoạt động liên tục.
\end{itemize}

\vspace{0.3cm}
\noindent\textbf{\textcolor{blue!80!black}{\faTasks \ 3. Tiêu chí đánh giá}}
\begin{itemize}[label=\textcolor{gray!70!black}{\faCaretRight}]
  \item \textbf{Sự hài lòng của người dùng:} Đánh giá mức độ thoải mái, mức độ cá nhân hóa và sự hài lòng tổng thể khi người dùng trải nghiệm các tính năng tìm kiếm, trò chuyện cùng AI trên nền tảng.
  \item \textbf{Tính hữu dụng:} Ứng dụng phải giải quyết được triệt để bài toán tìm kiếm chỗ ở. Giao diện cần trực quan, dễ thao tác và có luồng tương tác mượt mà phù hợp với nhiều nhóm đối tượng.
  \item \textbf{Độ chính xác của thông tin:} Đảm bảo chất lượng dữ liệu trả về. Danh sách địa điểm phải sát thực tế, và đặc biệt câu trả lời phân tích chi tiết từ AI phải chuẩn xác, tuyệt đối không được xảy ra hiện tượng Hallucination.
\end{itemize}


\section{System Overview}
\textbf{Hệ thống} là nền tảng tìm kiếm và quản lý lưu trú thông minh, kết hợp tra cứu truyền thống và sức mạnh Trí tuệ Nhân tạo. Dự án mang đến trải nghiệm đột phá qua giao diện hội thoại và tìm kiếm ngữ nghĩa.

\subsection{Overall Architecture}
\noindent Hệ thống được thiết kế theo mô hình \textbf{Modular Monolith} kết hợp với cơ chế điều phối tác tử AI. 

\begin{infobox}[\faQuestionCircle \ Tại sao chọn Modular Monolith?]
Lý do nhóm ưu tiên sử dụng \textbf{Modular Monolith} là nhằm giảm thiểu tối đa sự phức tạp trong quá trình triển khai và vận hành hệ thống ở giai đoạn nguyên mẫu. Với kiến trúc này, toàn bộ hệ thống được đóng gói chung để dễ quản lý, nhưng bên trong lại được phân tách nghiêm ngặt thành các phân hệ nghiệp vụ độc lập. Cách thiết kế này giúp mã nguồn luôn rõ ràng, ngăn chặn sự chồng chéo logic, tiết kiệm tài nguyên hệ thống, và đặc biệt là tạo tiền đề thuận lợi để dễ dàng bóc tách thành các dịch vụ độc lập dưới dạng Microservices trong tương lai khi dự án thực sự cần mở rộng quy mô.
\end{infobox}

\noindent Về tổng thể, kiến trúc bao gồm ba tầng chính:
\begin{itemize}[label=\textcolor{blue!70!black}{\faCubes}]
  \item \textbf{Tầng Giao diện:} Tập trung tối ưu hóa trải nghiệm người dùng, cung cấp bản đồ tương tác, không gian hỏi đáp cộng đồng và đặc biệt là khung trò chuyện AI với khả năng nhận luồng phản hồi liên tục theo thời gian thực.
  \item \textbf{Tầng Xử lý (Nghiệp vụ và AI):} Đóng vai trò là trung tâm điều phối, bao gồm các phân hệ quản lý người dùng, địa điểm và đánh giá. Điểm nhấn là phân hệ AI hoạt động như một cỗ máy điều phối quy trình công việc, chịu trách nhiệm phân tích ý định từ ngôn ngữ tự nhiên và tự động kích hoạt các công cụ tương ứng.
  \item \textbf{Tầng Dữ liệu:} Chịu trách nhiệm lưu trữ an toàn toàn bộ dữ liệu quan hệ của nền tảng. Đồng thời, tầng này cũng tích hợp khả năng lưu trữ và truy vấn các vector nhúng, làm cốt lõi cho tính năng tìm kiếm ngữ nghĩa và truy xuất dữ liệu tăng cường.
\end{itemize}

\subsection{Purpose and Approach}
Trong dự án này, Nhận diện mẫu đóng vai trò thiết yếu để định hình kiến trúc hệ thống. Thay vì giải quyết từng vấn đề một cách đơn lẻ, nhóm đã quan sát các trường hợp lặp lại trong quá trình vận hành, so sánh chúng, gom nhóm những điểm tương đồng và khái quát hóa thành các khuôn mẫu thiết kế hữu ích. Các phương pháp tổng quát như đối chiếu nhiều trường hợp, phân tích sự lặp lại và kiểm chứng bằng các kịch bản kiểm thử đã được áp dụng triệt để.

\noindent Việc lựa chọn cách tiếp cận này xuất phát từ đặc thù của hệ thống: phải tiếp nhận vô số câu lệnh tự nhiên đa dạng, vận hành nhiều luồng xử lý song song, tổng hợp thông tin từ nhiều nguồn và đối mặt với hàng loạt tình huống lỗi ngoại vi. Nếu xử lý từng trường hợp riêng lẻ, hệ thống sẽ trở nên rời rạc, thiếu đồng bộ và rất khó mở rộng. Nhờ đúc kết được 6 quy luật cốt lõi dưới đây, nhóm đã có cơ sở vững chắc để đưa ra các quyết định thiết kế kiến trúc chuẩn xác.

Mặc dù không theo một cấu trúc cố định, những thông tin này lại đóng vai trò tối quan trọng: chúng thường xuyên được sử dụng làm ngữ cảnh bổ trợ, cung cấp dữ liệu chi tiết cho Mô hình ngôn ngữ lớn nhằm tối ưu hóa và gia tăng độ chính xác cho câu trả lời.

\vspace{0.15cm}
\noindent Nhận thức được điều này, nhóm đánh giá việc thiết lập một khuôn mẫu chung để xử lý thống nhất các luồng ngôn ngữ tự do là một yêu cầu cấp thiết. Từ đó đã dẫn đến \textbf{ý tưởng then chốt}: tiến hành "số hóa" toàn bộ thông tin dạng ngôn ngữ tự nhiên.

\vspace{0.15cm}
\noindent Việc chuyển đổi câu chữ thành các dải số học không chỉ giúp máy tính hiểu được ý nghĩa sâu xa mà còn cho phép trích xuất chính xác các ngữ cảnh cần thiết nhất để phục vụ cho hệ thống xử lý.

\section{Abstraction}

\subsection{Role of Abstraction in System Architecture}

Đây là một nền tảng lưu trú du lịch thông minh, đòi hỏi sự phối hợp liên tục giữa nhiều thành phần có độ phức tạp cao như mô hình ngôn ngữ lớn LLM, dịch vụ bản đồ, giao tiếp dữ liệu địa điểm, cơ sở dữ liệu vector và các cơ chế xác thực. Nếu để hệ thống phụ thuộc trực tiếp vào đặc tả chi tiết của từng nhà cung cấp dịch vụ, toàn bộ kiến trúc sẽ trở nên cứng nhắc, khó mở rộng và dễ đứt gãy mỗi khi có sự thay đổi từ bên ngoài. Để giải quyết rào cản này, nhóm phát triển đã áp dụng mạnh mẽ tư duy trừu tượng hóa nhằm thiết lập các lớp khái niệm ổn định, che giấu hoàn toàn các chi tiết triển khai cụ thể phía sau. Tư duy này cho phép hệ thống giao tiếp với các dịch vụ cốt lõi dựa trên chức năng thiết yếu của chúng thay vì bị trói buộc vào một công nghệ hay nhà cung cấp nhất định.



\subsection{Abstracted Components}

Trong quá trình xây dựng kiến trúc hệ thống, tư duy trừu tượng hóa được áp dụng xuyên suốt trên nhiều khía cạnh khác nhau:

\vspace{0.15cm}
\noindent \textbf{\textcolor{blue!70!black}{1. Năng lực xử lý ngôn ngữ:}} Hệ thống không bị ràng buộc vào một nhà cung cấp AI cụ thể nào. Ở mức kiến trúc, LLM đơn thuần được xem như một tác nhân cung cấp khả năng thấu hiểu ngôn ngữ tự nhiên, phân tích ý định người dùng và tự động tạo ra phản hồi. Mọi sự khác biệt phức tạp về mô hình, định dạng giao tiếp API, tốc độ phản hồi hay cơ chế truyền dữ liệu liên tục đều được che giấu kỹ lưỡng. Nhờ vậy, luồng xử lý chính luôn được giữ ở trạng thái ổn định ngay cả khi tiến hành chuyển đổi nhà cung cấp hay nâng cấp mô hình.

\subsection{Architectural Benefits and Balance Level}

\noindent\textbf{Gia tăng sự linh hoạt:} Việc vận dụng triệt để Abstraction mang lại sự suy giảm đáng kể mức độ phụ thuộc chéo giữa các phân hệ, tạo không gian linh hoạt tối đa để thay thế hoặc luân chuyển nhà cung cấp. Sự phân tách này giúp hệ thống trở nên dễ kiểm thử, dễ bảo trì, và mở ra tiềm năng mở rộng không giới hạn trong tương lai. Hệ quả tất yếu là rủi ro kỹ thuật giảm thiểu tối đa, đặc biệt là khi các API bên ngoài thay đổi.

\vspace{0.15cm}
\noindent\textbf{Cân bằng mức độ trừu tượng:} Dù mang lại giá trị to lớn, nhóm phát triển ý thức được rằng trừu tượng hóa không phải là công cụ lạm dụng bừa bãi. Việc trừu tượng hóa quá mức sẽ biến hệ thống thành một cấu trúc mơ hồ, trong khi quá ít lại khiến logic nghiệp vụ bị gắn chặt với các nền tảng. Nhóm đã thận trọng lựa chọn điểm cân bằng: \textit{mọi lớp trừu tượng được tạo ra đều phải gắn liền với một định hướng kiến trúc rõ ràng}. Mỗi lớp trung gian chỉ tồn tại nếu nó thực sự giải quyết bài toán thay thế nhà cung cấp, chuẩn hóa dữ liệu, hoặc gia tăng khả năng mở rộng.

\section{System / Algorithm Design}

\subsection{Architectural Design Goals}
Hệ thống được thiết kế nhằm mang lại trải nghiệm khám phá các địa điểm lưu trú và ẩm thực trực quan thông qua bản đồ. Ngoài việc cung cấp thông tin, nền tảng tập trung mạnh vào sự tương tác của những người dùng đã xác thực thông qua các tính năng đánh giá, hỏi đáp, đánh dấu yêu thích và đặc biệt là cơ chế xác thực vị trí vật lý. Nền tảng kết nối hài hòa giữa dữ liệu địa điểm từ các nguồn bên ngoài và dữ liệu cộng đồng được lưu trữ bền vững bên trong. Trợ lý trí tuệ nhân tạo được tích hợp sâu vào hệ thống nhằm cung cấp các tư vấn chuẩn xác mà không làm gián đoạn hoặc cản trở thao tác của người dùng. Xét trên phương diện vận hành, hệ thống phải đảm bảo khả năng triển khai và bảo trì dễ dàng, phù hợp với năng lực quản lý của một nhóm phát triển sinh viên.

Để đạt được những mục tiêu trên, nhóm quyết định áp dụng kiến trúc Modular Monolith thay vì kiến trúc Microservices. Quyết định này giúp tối giản hóa đáng kể quy trình triển khai và môi trường local development, đồng thời giữ được một ranh giới database transaction chung duy nhất. Nhờ vậy, độ phức tạp trong vận hành được giảm tải tối đa, nhưng hệ thống vẫn đảm bảo được sự tách biệt rành mạch giữa các lĩnh vực nghiệp vụ thông qua cấu trúc logic module ở backend.

\subsection{Overall Architecture}
Hệ thống được phân tầng rõ rệt thành các lớp chuyên biệt, cấu thành một khối kiến trúc Modular Monolith hoàn chỉnh.

\textbf{Frontend:}
Sử dụng bộ công cụ React kết hợp Vite và Tailwind CSS. Lớp này đảm nhận toàn bộ giao diện tương tác, bao gồm bản đồ tìm kiếm, thẻ thông tin địa điểm, trang chi tiết địa điểm với khu vực hỏi đáp và đánh giá. Nó cũng quản lý hiển thị trạng thái có mặt vật lý, chức năng lưu địa điểm và trang hồ sơ cá nhân.

\textbf{Backend/Application Layer:}
Được xây dựng trên nền tảng NestJS. Lớp này đóng vai trò trung tâm điều phối mọi luồng nghiệp vụ. Nó xác thực danh tính người dùng, điều hướng các luồng tìm kiếm, đồng bộ hóa dữ liệu địa điểm bên ngoài vào cơ sở dữ liệu nội bộ. Ngoài ra, lớp này quản lý toàn bộ vòng đời của các bài đánh giá, chuỗi hỏi đáp, điều phối các phản hồi từ trí tuệ nhân tạo, xác minh vị trí vật lý, và lưu trữ danh sách yêu thích.

\textbf{Data Layer:}
Sử dụng hệ quản trị PostgreSQL kết hợp Prisma ORM. Đây là nơi lưu trữ bền vững mọi thông tin về người dùng, địa điểm, đánh giá, câu hỏi, câu trả lời, bản ghi trạng thái có mặt, danh sách yêu thích và cả lịch sử hội thoại của trợ lý thông minh.

\textbf{External Integrations:}
Hệ thống tận dụng các dịch vụ chuyên biệt từ bên ngoài nhằm tối ưu hiệu năng: Firebase phục vụ xác thực danh tính; SerpAPI cung cấp dữ liệu tìm kiếm địa điểm; Goong đảm nhận định vị tọa độ; và mô hình LLM Groq xử lý suy luận cho trợ lý thông minh. Các lớp này tương tác chặt chẽ thông qua các giao thức mạng bảo mật, trong đó lớp Ứng dụng đóng vai trò làm lá chắn trung gian.

\subsection{C4 Level 1: System Context}
Hệ thống phục vụ ba nhóm đối tượng chính: Khách vãng lai, Người dùng đã xác thực, và Người dùng tại chỗ. Khách vãng lai có quyền xem các thông tin địa điểm và nội dung hỏi đáp công khai. Khi đã xác thực, người dùng có thể đóng góp đánh giá, đặt câu hỏi, thảo luận, lưu địa điểm và báo cáo vị trí. Người dùng tại chỗ là những cá nhân đã được hệ thống xác minh tọa độ vật lý trùng khớp với địa điểm, giúp nâng cao độ tin cậy của các câu trả lời do họ đóng góp. Hệ thống tương tác với Firebase để xác thực, SerpAPI để tìm kiếm, Goong để xử lý tọa độ, và Groq để sinh văn bản thông minh, trong khi toàn bộ dữ liệu nội tại được lưu giữ tại PostgreSQL.

\subsection{Frontend Architecture}
Frontend đóng vai trò là tầng điều phối quy trình làm việc và tương tác. Thay vì nhồi nhét logic nghiệp vụ, giao diện chỉ gọi các dịch vụ máy chủ và phản ánh trạng thái dữ liệu. React Router quản lý điều hướng và bảo vệ các protected routes. Context API đảm trách duy trì trạng thái toàn cục như thông tin xác thực, dữ liệu bản đồ và ngữ cảnh hội thoại.

Trang Chủ đảm nhận việc khám phá qua bản đồ, áp dụng bộ lọc và render các thẻ địa điểm cùng điểm đánh dấu. Trang Chi tiết Địa điểm đi sâu vào thông tin không gian, hiển thị đánh giá, chuỗi hỏi đáp có gắn câu trả lời AI, nút xác nhận vị trí, nút lưu yêu thích và tích hợp luôn cửa sổ chat tư vấn chuyên biệt. Trang Đăng nhập là cửa ngõ duy nhất kết nối với hệ thống xác thực Firebase. Cuối cùng, Trang Hồ sơ cá nhân tóm tắt thông tin người dùng, trạng thái có mặt hiện tại và toàn bộ danh sách địa điểm đã lưu. Các thành phần giao diện như Bản đồ, Thẻ địa điểm, Khu vực Hỏi đáp được thiết kế độc lập, kết nối với nhau thông qua mạng lưới điều hướng và Error Boundary chặt chẽ.

\subsection{Data Architecture}
Cơ sở dữ liệu được chuẩn hóa nghiêm ngặt. Người dùng là trung tâm sở hữu vô số đánh giá, câu hỏi, câu trả lời, bản ghi vị trí, danh sách yêu thích và lịch sử hội thoại. Tương tự, Địa điểm là điểm neo cho các đánh giá, câu hỏi, vị trí và các tham chiếu hội thoại. Một câu hỏi chứa nhiều câu trả lời; các câu trả lời từ người dùng được liên kết nhân thân rõ ràng, trong khi các phản hồi được AI ghim lại tồn tại như những câu trả lời vô danh mang cờ hiệu hệ thống. Trạng thái vị trí lưu trữ chính xác thời điểm đến và đi. Bảng SavedPlace đóng vai trò cầu nối đa chiều giữa người dùng và địa điểm. \subsection{Key Runtime Workflows}

Trong số các luồng vận hành của hệ thống, ba workflow dưới đây thể hiện rõ nhất đóng góp về kiến trúc, xử lý dữ liệu và tích hợp trí tuệ nhân tạo. Các luồng hoạt động khác, dù có đóng góp vào kiến trúc hệ thống, nhưng nhóm sẽ không trình bày do độ phức tạp của các luồng đó không cao.

\subsubsection*{Workflow 1: Hybrid Search và Lazy Place Ingestion}

Workflow tìm kiếm bắt đầu từ truy vấn tự nhiên hoặc bộ lọc do người dùng cung cấp. \textit{Search Module} lựa chọn nguồn dữ liệu theo cấu hình runtime, có thể bao gồm cơ sở dữ liệu nội bộ, bộ fixture cục bộ hoặc nhà cung cấp bên ngoài. Các kết quả từ nhiều nguồn được chuẩn hóa về một cấu trúc địa điểm thống nhất, sau đó được xếp hạng và trả về giao diện để đồng bộ danh sách với bản đồ.

Kết quả bên ngoài chưa được ghi ngay vào PostgreSQL. Chỉ khi người dùng thực hiện một tương tác cần dữ liệu bền vững, chẳng hạn lưu, đánh giá, đặt câu hỏi hoặc check-in, hệ thống mới kích hoạt \textit{lazy ingestion}. Backend kiểm tra ánh xạ \texttt{PlaceSource}, tìm kiếm tương đồng theo tên, địa chỉ và tọa độ để tái sử dụng thực thể đã có; nếu không tìm thấy, hệ thống tạo một \texttt{Place} canonical cùng UUID nội bộ. Cách tiếp cận này tránh nhập dư thừa toàn bộ dữ liệu từ nhà cung cấp nhưng vẫn bảo đảm mọi dữ liệu cộng đồng được liên kết nhất quán.

\subsubsection*{Workflow 2: AI Assistant theo Workflow có Xác nhận}

AI Assistant không xử lý mọi yêu cầu như một hội thoại tự do. Khi nhận tin nhắn, \textit{Task Router} phân loại ý định thành tìm kiếm, so sánh, phân tích một địa điểm hoặc hội thoại thông thường, đồng thời trích xuất các tham số liên quan. Với các tác vụ có ảnh hưởng đến kết quả nghiệp vụ, frontend hiển thị wizard để người dùng bổ sung tiêu chí và xác nhận trước khi thực thi.

Sau xác nhận, \textit{Workflow Engine} chạy chuỗi công cụ xác định được khai báo trong registry. Workflow tìm kiếm lần lượt thực hiện hybrid search, geocode điểm neo và xếp hạng; workflow phân tích tổng hợp metadata, điểm xuất phát, ước lượng di chuyển và các địa điểm lân cận; workflow so sánh sử dụng tập địa điểm đã được người dùng tag. Kết quả công cụ được \textit{Response Composer} chuyển thành câu trả lời tự nhiên hoặc payload có cấu trúc. \subsubsection*{Workflow 3: Q\&A Địa điểm kết hợp AI và Cộng đồng}

Thread vừa tạo được trả về ngay để giao diện cập nhật mà không phải chờ mô hình ngôn ngữ. Việc sinh câu trả lời AI diễn ra bất đồng bộ ở nền: hệ thống tập hợp metadata địa điểm, tiện ích, rating và các trích đoạn review, sau đó yêu cầu LLM tạo câu trả lời dựa trên bằng chứng hiện có.

\subsection{Security and Access Control}
Bảo mật hệ thống ủy thác cho Firebase ở client-side. Bất kỳ tác vụ cập nhật dữ liệu nào cũng phải đính kèm thẻ xác thực Firebase để NestJS xác thực qua Admin SDK. Các thao tác đọc thông tin công cộng được nới lỏng nhằm tối ưu trải nghiệm, nhưng mọi thao tác ghi, đặc biệt là quyền xóa đánh giá hay xóa câu hỏi, đều bị kiểm tra quyền sở hữu cực kỳ gắt gao. Khi một câu hỏi bị xóa, hệ thống kích hoạt cơ chế cascade delete dọn dẹp sạch sẽ các câu trả lời liên quan, đảm bảo không lưu lại rác dữ liệu. Trạng thái vị trí cũng có thể tự do xóa bỏ để bảo vệ quyền ẩn danh.

\subsection{Runtime Configuration and Environment Strategy}
Hệ thống áp dụng chiến lược Runtime Configuration thông qua Dependency Injection của NestJS để chuyển đổi mượt mà giữa môi trường Testing và Production. Việc branching logic được hạn chế tối đa; thay vào đó, toàn bộ cấu hình được trừu tượng hóa và nạp động từ các tệp \texttt{.env} tương ứng. 

Đặc biệt ở chế độ Testing Mode, cơ chế phát triển được tối ưu hóa như sau:
\begin{itemize}
    \item \textbf{Mocking External APIs:} Thay vì gửi request thực tế tốn chi phí và phụ thuộc mạng lưới, các dịch vụ ngoại vi (SerpAPI, Goong, Groq LLM) được hệ thống tự động inject các các Mock Providers trả về kết quả định trước. 
    \item \textbf{Isolated Test Database:} Chế độ kiểm thử sử dụng một schema cơ sở dữ liệu hoàn toàn tách biệt. Dữ liệu được teardown (xóa sạch) sau mỗi chu kỳ chạy E2E Test, đảm bảo tính tất định tuyệt đối cho các kịch bản kiểm thử.
    \item \textbf{Ngắt lưu trữ rác:} Các chức năng ghi log hội thoại AI, xác thực Firebase thực tế hay đồng bộ dữ liệu tĩnh được vô hiệu hóa hoặc thay thế bằng cơ chế nội bộ để tăng tốc độ chạy test.
\end{itemize}

Cơ chế này cho phép các kỹ sư liên tục chạy unit test và e2e test ở môi trường cục bộ một cách độc lập, ổn định và không lo ngại giới hạn rate-limit của API bên ngoài.

\subsection{Current Limitations and Future Improvements}
Kiến trúc bộc lộ một số giới hạn nhất định trong phạm vi đồ án hiện tại. Tính năng tinh chỉnh bộ lọc tìm kiếm nâng cao theo phân loại địa điểm vẫn đang trong giai đoạn hoàn thiện. Luồng điều hướng và giao diện chỉ đường chưa đạt được sự mượt mà tối đa. Hệ thống cũng chịu sự ràng buộc bởi các dịch vụ cung cấp dữ liệu tìm kiếm, định vị và xử lý ngôn ngữ bên ngoài. Hơn nữa, chất lượng tư vấn của AI phụ thuộc tuyến tính vào lượng dữ liệu thực tế thu thập được; do đó, cơ chế kiểm duyệt và xếp hạng bình chọn cho các câu hỏi sẽ cần được bổ sung để nâng cấp chất lượng tri thức cộng đồng.

\subsection{Summary}
Kiến trúc hệ thống thể hiện sự rạch ròi tuyệt đối giữa việc render giao diện, xử lý luồng nghiệp vụ cốt lõi, lưu trữ dữ liệu bền vững và giao tiếp ngoại vi. Việc ứng dụng linh hoạt mô hình Modular Monolith giúp nhóm phát triển đạt được sự cân bằng hoàn hảo giữa tính dễ bảo trì và sự tinh giản trong vận hành. Nhờ những cơ chế đột phá như xác thực tọa độ vị trí vật lý và quy trình đồng bộ dữ liệu ngoại vi theo thời gian thực, nền tảng không chỉ là một công cụ tìm kiếm đơn thuần mà đã thực sự chuyển mình thành một mạng lưới tri thức cộng đồng an toàn, đáng tin cậy và mang đậm tính cá nhân hóa sâu sắc.
\section{Implementation}

\subsection{Implementation Strategy}
Thay vì lập trình tự phát theo từng tính năng lẻ tẻ, nhóm tiếp cận việc hiện thực hóa bằng cách bám sát luồng nghiệp vụ cốt lõi đã thiết kế. Mục tiêu đầu tiên là xây dựng nhanh một bản MVP (Minimum Viable Product) để kiểm chứng ý tưởng. Các thành phần được chia nhỏ thành những module độc lập, cho phép mỗi thành viên phát triển và kiểm thử cục bộ trước khi tích hợp vào hệ thống chung. \subsection{Quản lý ngữ cảnh giao tiếp cho mô hình ngôn ngữ}
\label{sec:llm_context}
Để mô hình ngôn ngữ lớn (LLM) có thể thực hiện trọn vẹn vai trò của một trợ lý du lịch ảo, rào cản lớn nhất không nằm ở sức mạnh suy luận của mô hình, mà là cách thiết kế luồng cung cấp ngữ cảnh đúng, đủ và liên tục. Nhóm đã xây dựng một cơ chế quản lý ngữ cảnh đa tầng xuyên suốt từ giao diện người dùng đến máy chủ xử lý:

\begin{itemize}
    \item \textbf{Đóng gói trạng thái giao diện trực tiếp:} Khi người dùng nhắn tin, dữ liệu gửi đi không chỉ là văn bản thô. Tầng giao diện tự động thu thập các địa điểm đang được chọn, luồng công việc hiện tại, và các giới hạn ngân sách. Các thông tin này được đính kèm vào khối dữ liệu tải lên. Điều này giúp LLM có khả năng "nhìn" thấy bản đồ, hiểu được người dùng đang quan tâm đến địa điểm nào mà không cần họ phải gọi tên cụ thể.
    \item \textbf{Trích xuất ý định bằng luồng AI song song:} Thay vì sử dụng các biểu thức chính quy để bóc tách từ khóa tìm kiếm, hệ thống định hướng đầu vào bằng một lời gọi LLM phụ trợ siêu tốc. Lời gọi này có nhiệm vụ duy nhất là phân tích văn bản và trả về một cấu trúc dữ liệu JSON bao gồm: cờ xác nhận tìm kiếm, vị trí, phân loại (chỗ ở, ăn uống, tham quan) và ngân sách. Nhờ vậy, hệ thống có thể nhận diện yêu cầu di chuyển bản đồ từ những câu hội thoại tự nhiên nhất.
    \item \textbf{Tiêm dữ liệu động chống hiện tượng Hallucination:} Để trả lời chuyên sâu về một địa điểm cụ thể, hệ thống tuyệt đối không để LLM tự suy diễn dựa trên kiến thức cũ. Dịch vụ xử lý trò chuyện sẽ tự động trích xuất từ cơ sở dữ liệu một khối thông tin chứa các dữ kiện xác thực: nguồn cung cấp, danh mục, điểm đánh giá trung bình, các tiện ích, và đặc biệt là tối đa 5 trích đoạn đánh giá thực tế nổi bật nhất. Chỉ thị gốc của hệ thống (System Prompt) được thiết lập nghiêm ngặt: yêu cầu LLM chỉ sử dụng các bằng chứng vừa được cung cấp, trả lời bằng đại từ xưng hô gần gũi ("mình" - "bạn"), và thẳng thắn từ chối nếu thông tin không đủ để đưa ra kết luận.
    \item \textbf{Quản lý bộ nhớ hội thoại dạng cửa sổ trượt:} Nhằm tối ưu hóa giới hạn ngữ cảnh đầu vào và giảm thiểu chi phí API, lịch sử trò chuyện được kiểm soát chặt chẽ. Hệ thống duy trì một bộ nhớ đệm cục bộ với thời gian sống là 3600 giây, và tự động cắt xén chỉ giữ lại đúng 20 tin nhắn gần nhất. Đồng thời, cấu hình hệ thống cho phép đồng bộ hóa dữ liệu này vào cơ sở dữ liệu PostgreSQL một cách phi đồng bộ, đảm bảo tính liên tục của các phiên hội thoại khi tải lại trang mà không làm chậm luồng phản hồi chính.
\end{itemize}

    \item \textbf{Khả năng mở rộng không tạo hồi quy:} Đợt bổ sung 16 tập tin kiểm thử mới nhất (bao gồm cả phân tích không gian, cấu hình runtime, hệ thống AI Orchestrator và Frontend Router) đã \textbf{vượt qua 100\% các ca thử nghiệm} mà không làm phá vỡ bất kỳ logic lõi nào. Điều này chứng minh cấu trúc của Backend đang làm rất tốt việc cô lập phạm vi ảnh hưởng.
    \item \textbf{Độ bao phủ của hệ thống thông minh:} Toàn bộ hệ sinh thái truy xuất tăng cường, phân tích ngôn ngữ tự nhiên, và luồng thực thi trí tuệ nhân tạo đều hoạt động hoàn hảo dưới môi trường giả lập. Máy trạng thái phân giải ý định người dùng không bị sai lệch ngay cả khi nhận vào các chuỗi ngôn ngữ tự nhiên phức tạp.
    \item \textbf{Tốc độ phản hồi:} Hoàn tất hơn 570 bài kiểm tra cho cả hai đầu hệ thống trong vòng xấp xỉ hai phút là một chỉ số hiệu suất ấn tượng, tạo tiền đề lý tưởng cho việc thiết lập các luồng tự động hóa triển khai nghiêm ngặt sau này.
\end{itemize}

\subsubsection{Điểm hạn chế}
Toàn bộ các ca thất bại hiện tại (33 lỗi ở backend và 11 lỗi ở frontend) đều là \textbf{các lỗi tồn đọng từ các đợt tái cấu trúc trước}, hoàn toàn không phải lỗi logic nghiệp vụ cốt lõi mới phát sinh. Phân tích nguyên nhân gốc rễ chỉ ra ba vấn đề chính:

\begin{enumerate}
    \item \textbf{Mất đồng bộ cơ chế Dependency Injection:} Hơn 9 ca lỗi ở Backend xuất phát từ việc các bộ kiểm thử Controller sử dụng hệ thống Guard bảo mật nhưng kịch bản kiểm thử lại chưa cung cấp dịch vụ cấu hình vào trong khối khởi tạo của module kiểm thử.
    \item \textbf{Đối tượng giả lập không hoàn chỉnh:} Tại Frontend, các dịch vụ liên quan trực tiếp đến cơ sở dữ liệu thời gian thực thất bại do đối tượng giả lập không phản ánh đúng chữ ký hàm của SDK mới nhất. Tương tự ở Backend, một số luồng thao tác với trí tuệ nhân tạo bị gián đoạn trong môi trường kiểm thử do việc giả lập thiếu các thuộc tính bắt buộc.
    \item \textbf{Mất đồng bộ chuẩn định dạng dữ liệu và mã nguồn cũ:} Sự vắng mặt của các tệp khai báo kiểu dữ liệu và sự tồn tại của các tập tin kiểm thử cũ (sử dụng API cũ đã bị xóa bỏ) gây ra các lỗi biên dịch hoặc lỗi tham chiếu. Ngoài ra, một component hiển thị tại frontend đang gặp lỗi do bộ công cụ kiểm thử cố gắng tìm một chuỗi văn bản thuần thay vì sử dụng nhãn nhận dạng hỗ trợ tiếp cận vốn đã được áp dụng để tối ưu giao diện.
\end{enumerate}

\textbf{Kết luận rà soát:} 
Hệ thống lõi hoạt động ổn định và có mức độ hoàn thiện cao, sẵn sàng cho việc đóng gói và triển khai thực tế. Việc dọn dẹp các ca thất bại không đòi hỏi thiết kế lại thuật toán, mà thuần túy là quá trình đồng bộ hóa cấu hình môi trường kiểm thử với cấu trúc mã nguồn đã được nâng cấp.

\subsection{Conclusion}
Nhìn lại toàn bộ hành trình thực hiện đồ án, nhóm đã nghiên cứu, thiết kế và phát triển thành công một hệ thống tìm kiếm và gợi ý địa điểm lưu trú tích hợp Trí tuệ Nhân tạo. Khởi điểm từ việc áp dụng bốn trụ cột của Tư duy Máy tính để phân rã bài toán, nhóm đã chuyển hóa thành công các yêu cầu trừu tượng thành một hệ thống thực tế hoàn chỉnh. Việc áp dụng kiến trúc Modular Monolith kết hợp cùng các công nghệ hiện đại như NestJS, React, cơ sở dữ liệu vector pgvector, và hệ thống truy xuất thông tin RAG thông qua Groq LLM đã chứng minh được tính hiệu quả và khả năng mở rộng của nền tảng. 

Sản phẩm cuối cùng không chỉ đáp ứng các chức năng cốt lõi đã đề ra---như tìm kiếm bằng ngôn ngữ tự nhiên, so sánh địa điểm và trích xuất thông tin tự động---mà còn đảm bảo tính nhất quán về mặt kiến trúc phần mềm và quy trình kiểm thử.

\subsection{Team Reflections and Lessons Learned}
Đồ án này là một thách thức lớn nhưng đồng thời cũng là một cơ hội quý báu để các thành viên trong nhóm cọ xát với môi trường phát triển phần mềm chuyên nghiệp. Từ quá trình thực tiễn, nhóm đã đúc kết được nhiều bài học sâu sắc:

\begin{itemize}
    \item \textbf{Tầm quan trọng của khâu thiết kế kiến trúc:} Việc thay đổi chiến lược từ Microservices sang Modular Monolith ở giai đoạn đầu đã dạy cho nhóm bài học đắt giá về việc lựa chọn kiến trúc phù hợp với quy mô dự án và năng lực đội ngũ. Một thiết kế tốt sẽ giúp giảm thiểu nợ kỹ thuật ở các giai đoạn sau.
    \item \textbf{Cân bằng giữa công nghệ mới và tính ổn định:} Việc ứng dụng các công nghệ mới như pgvector hay mô hình ngôn ngữ lớn (LLM) đòi hỏi sự kiên nhẫn trong việc nghiên cứu tài liệu. Nhóm nhận ra rằng AI không phải là "phép thuật" giải quyết mọi vấn đề, mà cần được tích hợp có kiểm soát thông qua Prompt Engineering chặt chẽ để tránh hiện tượng Hallucination.
    \item \textbf{Kỷ luật trong làm việc nhóm và quản lý mã nguồn:} Quá trình vận hành dự án với nhiều thành viên đã chứng minh giá trị của việc tuân thủ các quy chuẩn Git, kiểm thử liên tục, và giao tiếp minh bạch. Sự hỗ trợ từ các công cụ AI Agents (như Antigravity CLI) cũng đóng vai trò xúc tác lớn, giúp nhóm tối ưu hóa hiệu suất lập trình.
\end{itemize}

Nhận xét chung về cá nhân và tập thể, mặc dù dự án vẫn còn một số điểm có thể tối ưu hóa thêm về mặt hiệu năng xử lý song song, nhưng kết quả đạt được đã vượt qua sự kỳ vọng ban đầu. Qua đồ án này, mỗi cá nhân trong nhóm đều đã tự nâng tầm tư duy hệ thống và hoàn thiện kỹ năng giải quyết bài toán cốt lõi.

\subsection{Future Development Directions}
Để phát triển dự án từ một nguyên mẫu thành một sản phẩm thương mại hoàn chỉnh, nhóm đề xuất các hướng cải tiến sau:
\begin{itemize}
    \item \textbf{Cải tiến mô hình RAG và Fine-tuning LLM:} Tối ưu hóa chiến lược truy xuất Semantic Search, áp dụng các kỹ thuật phân mảnh văn bản thông minh hơn và tiến tới fine-tune một mô hình LLM chuyên biệt cho lĩnh vực du lịch nhằm tăng tốc độ phản hồi.
    \item \textbf{Mở rộng sang kiến trúc Microservices thực thụ:} Khi hệ thống cần phục vụ lượng người dùng lớn hơn, việc tách các module (như AI Engine, Authentication, Core Services) thành các dịch vụ độc lập sẽ được cân nhắc triển khai cùng với hệ thống điều phối container.
    \item \textbf{Tối ưu hóa UI/UX và phân tích dữ liệu:} Nâng cấp trải nghiệm người dùng với các giao diện cá nhân hóa sâu hơn, tích hợp hệ thống tracking hành vi để hệ thống có khả năng gợi ý địa điểm lưu trú một cách chủ động.
\end{itemize}
