\section{Idea}

\subsection{Đặt vấn đề}
Hiện nay, việc tìm kiếm một nơi lưu trú phù hợp khi đi du lịch không hề đơn giản. Người dùng thường phải cân nhắc nhiều yếu tố như giá cả, vị trí, tiện nghi, chất lượng dịch vụ và các đánh giá từ những khách hàng trước đó. Quá trình này vẫn còn gặp nhiều khó khăn với các \textbf{điểm đau} cốt lõi:

\begin{itemize}[label=\textcolor{blue!70!black}{\faTimesCircle}]
  \item \textbf{Thông tin phân mảnh và Rời rạc:} Dữ liệu về khách sạn, homestay rải rác trên nhiều nền tảng, khiến người dùng mất nhiều thời gian tổng hợp và so sánh.
  \item \textbf{Bộ lọc cứng nhắc và Thiếu cá nhân hoá:} Các công cụ truyền thống chỉ cho phép lọc theo các tùy chọn có sẵn, không hỗ trợ tìm kiếm linh hoạt dựa trên nhu cầu tự nhiên và đặc thù của người dùng.
  \item \textbf{Độ tin cậy của dữ liệu thấp:} Đánh giá từ người dùng trước thường mang tính chủ quan, có thể bị thao túng bằng các đánh giá giả mạo và thiếu tính cập nhật thực tế tại thời điểm hiện tại.
  \item \textbf{Quá tải thông tin và Tốn thời gian:} Người dùng mất quá nhiều thời gian để đọc và đối chiếu thông tin từ nhiều nguồn trước khi có thể đưa ra quyết định chính xác.
\end{itemize}

\begin{infobox}[\faLightbulb \ Giải pháp đề xuất]
Nhóm xây dựng hệ thống tìm kiếm và gợi ý lưu trú thông minh, cho phép người dùng mô tả nhu cầu bằng \textbf{ngôn ngữ tự nhiên}. Hệ thống sẽ phân tích yêu cầu, tìm kiếm các địa điểm phù hợp, đánh giá, xếp hạng và đưa ra những gợi ý trực quan, giúp người dùng ra quyết định nhanh chóng.
\end{infobox}

\subsection{Tầm nhìn ứng dụng}
Dự án không chỉ nhằm xây dựng một nền tảng thương mại thông thường, mà định hình lại cách người dùng tương tác với dữ liệu. Thông qua việc ứng dụng AI và kỹ thuật \textbf{RAG}, hệ thống kỳ vọng tạo ra một tiền đề cho tương lai, nơi AI đóng vai trò như một người điều phối đáng tin cậy.

\subsection{Người dùng mục tiêu}
\begin{itemize}[label=\textcolor{cyan!70!black}{\faUserCheck}]
  \item \textbf{Khách du lịch tự túc:} Cần tìm kiếm nhanh chóng, so sánh giá để tối ưu hóa trải nghiệm.
  \item \textbf{Người dùng On-site:} Đang lưu trú tại địa điểm, sẵn sàng đóng góp thông tin thực tế và trả lời câu hỏi cộng đồng.
  \item \textbf{Người ưu tiên cá nhân hóa:} Thích tương tác với AI thay vì tốn hàng giờ với các bộ lọc phức tạp.
\end{itemize}


\section{Problem Analysis}

\subsection{Hành trình Người dùng}
Thay vì tiếp cận theo hướng kỹ thuật khô khan, nhóm thiết kế hệ thống thông qua chuỗi câu hỏi dẫn dắt từ góc nhìn người dùng:

\begin{itemize}[label=\textcolor{blue!70!black}{\faQuestionCircle}, leftmargin=0.5cm]
  \item \textbf{Người dùng sẽ làm gì với ứng dụng của tôi?} \\
  $\rightarrow$ Người dùng cần tìm địa điểm lưu trú.
  
  \item \textbf{Người dùng có thể tìm kiếm địa điểm lưu trú như thế nào?} \\
  $\rightarrow$ Thiết kế giao diện tìm kiếm hoặc phương thức tìm kiếm linh hoạt.

  \item \textbf{Vậy làm sao để phương thức tìm kiếm có thể cho ra kết quả tốt?} \\
  $\rightarrow$ Chúng ta sẽ đánh giá trên các tiêu chí, như vậy chúng ta cần trích xuất ý định của người dùng.

  \item \textbf{Vậy việc trích xuất nên được thực hiện thế nào?} \\
  $\rightarrow$ Có thể cho phép người dùng chọn hoặc thông qua tác tử AI được cài đặt.

  \item \textbf{Nếu người dùng sau khi đã có các địa điểm phù hợp với nhu cầu của họ, làm cách nào để hệ thống có thể giúp người dùng ra quyết định cuối cùng như thế nào?} \\
  $\rightarrow$ Có thể cài đặt tính năng giúp người dùng so sánh các địa điểm, hỗ trợ ra quyết định cuối.

  \item \textbf{Vậy nếu người dùng muốn tìm hiểu sâu hơn về một địa điểm thì sao?} \\
  $\rightarrow$ Cài đặt tính năng AI đưa ra phân tích chi tiết về một địa điểm cụ thể, cho phép người dùng lựa chọn các tiêu chí để phân tích.

  \item \textbf{Việc sử dụng LLM có thể không đạt được độ chính xác cao, liệu nguồn thông tin nào uy tín hơn không?} \\
  $\rightarrow$ Sử dụng nguồn thông tin từ chính người dùng, có thể áp dụng chính sách 'người dùng onsite' để tăng độ tin cậy.

  \item \textbf{Vậy với các tính năng như trên, có thể tối giản giao diện để người dùng có thể tương tác nhanh chóng và dễ dàng, phù hợp với nhiều người dùng ở các độ tuổi, trình độ khác nhau hay không?} \\
  $\rightarrow$ Sử dụng LLM đóng vai trò điều phối trung gian, người dùng sẽ tương tác trực tiếp với LLM, LLM sẽ trích xuất ý định người dùng và tiến hành các yêu cầu một cách tự động.
\end{itemize}

\subsection{Phân rã Bài toán Tính toán}
Từ nhu cầu thực tế, hệ thống được phân rã thành các module kỹ thuật chuyên biệt:
\begin{itemize}[label=\textcolor{blue!60}{\faCogs}]
  \item \textbf{Intent Parsing:} LLM chuyển đổi ngôn ngữ tự nhiên thành JSON cấu trúc.
  \item \textbf{Data Aggregation:} Truy xuất đồng thời DB nội bộ và API đối tác.
  \item \textbf{Semantic Search:} Số hóa văn bản thành vector nhúng (với pgvector).
  \item \textbf{Workflow Engine:} Xây dựng khung làm việc cứng kiểm soát AI.
\end{itemize}

\subsection{Ràng buộc, Giả định và Tiêu chí}
Để đảm bảo tính khả thi trong phạm vi đồ án, hệ thống được thiết kế và đánh giá dựa trên các khía cạnh rõ ràng sau:

\vspace{0.3cm}
\noindent\textbf{\textcolor{blue!80!black}{\faLink \ 1. Ràng buộc}}
\begin{itemize}[label=\textcolor{gray!70!black}{\faCaretRight}]
  \item \textbf{Về mặt kiến trúc:} Lựa chọn mô hình \textit{Modular Monolith} để giảm thiểu chi phí quản lý và vận hành trong giai đoạn MVP, đồng thời vẫn giữ được tính phân tách cao giúp dễ dàng mở rộng sang Microservices sau này.
  \item \textbf{Về mặt tài nguyên:} Hệ thống chuyển giao toàn bộ tải tính toán AI nặng nề sang các API của bên thứ ba (như Groq). Nhờ đó, hệ thống lõi có thể chạy cục bộ mượt mà trên phần cứng cơ bản không yêu cầu GPU.
  \item \textbf{Về phạm vi hỗ trợ:} Ứng dụng ở giai đoạn này tập trung vào thị trường Việt Nam. Hệ thống tập trung xác định các địa điểm lưu trú trong nước, yêu cầu AI phải hỗ trợ tiếng Việt tốt và hiểu rõ ngữ cảnh văn hóa địa phương.
\end{itemize}

\vspace{0.3cm}
\noindent\textbf{\textcolor{blue!80!black}{\faLightbulb \ 2. Giả định}}
\begin{itemize}[label=\textcolor{gray!70!black}{\faCaretRight}]
  \item Khả năng lập luận, phân tích ngữ nghĩa và đưa ra quyết định của Mô hình ngôn ngữ lớn là yếu tố khách quan, tính chính xác phụ thuộc vào API cung cấp chứ không bị giới hạn bởi hệ thống nội bộ.
  \item Dữ liệu thu thập từ các nhà cung cấp bên thứ ba (Goong, SerpAPI) luôn đảm bảo độ tin cậy về mặt nội dung và ổn định về thời gian phản hồi (uptime).
\end{itemize}

\vspace{0.3cm}
\noindent\textbf{\textcolor{blue!80!black}{\faTasks \ 3. Tiêu chí đánh giá}}
\begin{itemize}[label=\textcolor{gray!70!black}{\faCaretRight}]
  \item \textbf{Sự hài lòng của người dùng:} Đánh giá mức độ thoải mái, mức độ cá nhân hóa và sự hài lòng tổng thể khi người dùng trải nghiệm các tính năng tìm kiếm, trò chuyện cùng AI trên nền tảng.
  \item \textbf{Tính hữu dụng:} Ứng dụng phải giải quyết được triệt để bài toán tìm kiếm chỗ ở. Giao diện cần trực quan, dễ thao tác và có luồng tương tác mượt mà phù hợp với nhiều nhóm đối tượng.
  \item \textbf{Độ chính xác của thông tin:} Đảm bảo chất lượng dữ liệu trả về. Danh sách địa điểm phải sát thực tế, và đặc biệt câu trả lời phân tích chi tiết từ AI phải chuẩn xác, tuyệt đối không được xảy ra tình trạng ảo giác.
\end{itemize}


\section{System Overview}
\textbf{Hệ thống} là nền tảng tìm kiếm và quản lý lưu trú thông minh, kết hợp tra cứu truyền thống và sức mạnh Trí tuệ Nhân tạo. Dự án mang đến trải nghiệm đột phá qua giao diện hội thoại và tìm kiếm ngữ nghĩa.

\subsection{Kiến trúc tổng quan}
\noindent Hệ thống được thiết kế theo mô hình \textbf{Modular Monolith} kết hợp với cơ chế điều phối tác tử AI. 

\begin{infobox}[\faQuestionCircle \ Tại sao chọn Modular Monolith?]
Lý do nhóm ưu tiên sử dụng \textbf{Modular Monolith} là nhằm giảm thiểu tối đa sự phức tạp trong quá trình triển khai và vận hành hệ thống ở giai đoạn nguyên mẫu. Với kiến trúc này, toàn bộ hệ thống được đóng gói chung để dễ quản lý, nhưng bên trong lại được phân tách nghiêm ngặt thành các phân hệ nghiệp vụ độc lập. Cách thiết kế này giúp mã nguồn luôn rõ ràng, ngăn chặn sự chồng chéo logic, tiết kiệm tài nguyên hệ thống, và đặc biệt là tạo tiền đề thuận lợi để dễ dàng bóc tách thành các dịch vụ độc lập (Microservices) trong tương lai khi dự án thực sự cần mở rộng quy mô.
\end{infobox}

\noindent Về tổng thể, kiến trúc bao gồm ba tầng chính:
\begin{itemize}[label=\textcolor{blue!70!black}{\faCubes}]
  \item \textbf{Tầng Giao diện:} Tập trung tối ưu hóa trải nghiệm người dùng, cung cấp bản đồ tương tác, không gian hỏi đáp cộng đồng và đặc biệt là khung trò chuyện AI với khả năng nhận luồng phản hồi liên tục theo thời gian thực.
  \item \textbf{Tầng Xử lý (Nghiệp vụ và AI):} Đóng vai trò là trung tâm điều phối, bao gồm các phân hệ quản lý người dùng, địa điểm và đánh giá. Điểm nhấn là phân hệ AI hoạt động như một cỗ máy điều phối quy trình công việc, chịu trách nhiệm phân tích ý định từ ngôn ngữ tự nhiên và tự động kích hoạt các công cụ tương ứng.
  \item \textbf{Tầng Dữ liệu:} Chịu trách nhiệm lưu trữ an toàn toàn bộ dữ liệu quan hệ của nền tảng. Đồng thời, tầng này cũng tích hợp khả năng lưu trữ và truy vấn các vector nhúng, làm cốt lõi cho tính năng tìm kiếm ngữ nghĩa và truy xuất dữ liệu tăng cường.
\end{itemize}

\subsection{Mục đích và phương pháp tiếp cận}
Trong dự án này, Nhận diện mẫu đóng vai trò thiết yếu để định hình kiến trúc hệ thống. Thay vì giải quyết từng vấn đề một cách đơn lẻ, nhóm đã quan sát các trường hợp lặp lại trong quá trình vận hành, so sánh chúng, gom nhóm những điểm tương đồng và khái quát hóa thành các khuôn mẫu thiết kế hữu ích. Các phương pháp tổng quát như đối chiếu nhiều trường hợp, phân tích sự lặp lại và kiểm chứng bằng các kịch bản kiểm thử (test cases) đã được áp dụng triệt để.

\noindent Việc lựa chọn cách tiếp cận này xuất phát từ đặc thù của hệ thống: phải tiếp nhận vô số câu lệnh tự nhiên đa dạng, vận hành nhiều luồng xử lý song song, tổng hợp thông tin từ nhiều nguồn và đối mặt với hàng loạt tình huống lỗi ngoại vi. Nếu xử lý từng trường hợp riêng lẻ, hệ thống sẽ trở nên rời rạc, thiếu đồng bộ và rất khó mở rộng. Mặc dù không theo một cấu trúc cố định, những thông tin này lại đóng vai trò tối quan trọng: chúng thường xuyên được sử dụng làm ngữ cảnh bổ trợ, cung cấp dữ liệu chi tiết cho Mô hình ngôn ngữ lớn nhằm tối ưu hóa và gia tăng độ chính xác cho câu trả lời.

\vspace{0.15cm}
\noindent Nhận thức được điều này, nhóm đánh giá việc thiết lập một khuôn mẫu chung để xử lý thống nhất các luồng ngôn ngữ tự do là một yêu cầu cấp thiết. Từ đó đã dẫn đến \textbf{ý tưởng then chốt}: tiến hành "số hóa" toàn bộ thông tin dạng ngôn ngữ tự nhiên.

\vspace{0.15cm}

\section{Abstraction}

\subsection{Vai trò của Abstraction trong kiến trúc hệ thống}

Đây là một nền tảng lưu trú du lịch thông minh, đòi hỏi sự phối hợp liên tục giữa nhiều thành phần có độ phức tạp cao như mô hình ngôn ngữ lớn (LLM), dịch vụ bản đồ, giao tiếp dữ liệu địa điểm, cơ sở dữ liệu vector và các cơ chế xác thực. Nếu để hệ thống phụ thuộc trực tiếp vào đặc tả chi tiết của từng nhà cung cấp dịch vụ, toàn bộ kiến trúc sẽ trở nên cứng nhắc, khó mở rộng và dễ đứt gãy mỗi khi có sự thay đổi từ bên ngoài. Để giải quyết rào cản này, nhóm phát triển đã áp dụng mạnh mẽ tư duy trừu tượng hóa (Abstraction) nhằm thiết lập các lớp khái niệm ổn định, che giấu hoàn toàn các chi tiết triển khai cụ thể phía sau. Tư duy này cho phép hệ thống giao tiếp với các dịch vụ cốt lõi dựa trên chức năng thiết yếu của chúng thay vì bị trói buộc vào một công nghệ hay nhà cung cấp nhất định.



\subsection{Các thành phần được trừu tượng hóa}

Trong quá trình xây dựng kiến trúc hệ thống, tư duy trừu tượng hóa được áp dụng xuyên suốt trên nhiều khía cạnh khác nhau:

\vspace{0.15cm}
\noindent \textbf{\textcolor{blue!70!black}{1. Năng lực xử lý ngôn ngữ (LLM):}} Hệ thống không bị ràng buộc vào một nhà cung cấp AI cụ thể nào. Ở mức kiến trúc, LLM đơn thuần được xem như một tác nhân cung cấp khả năng thấu hiểu ngôn ngữ tự nhiên, phân tích ý định người dùng và tự động tạo ra phản hồi. Mọi sự khác biệt phức tạp về mô hình, định dạng giao tiếp API, tốc độ phản hồi hay cơ chế truyền dữ liệu liên tục (streaming) đều được che giấu kỹ lưỡng. Nhờ vậy, luồng xử lý chính luôn được giữ ở trạng thái ổn định ngay cả khi tiến hành chuyển đổi nhà cung cấp hay nâng cấp mô hình.

\subsection{Lợi ích kiến trúc và mức độ cân bằng}

\noindent\textbf{Gia tăng sự linh hoạt:} Việc vận dụng triệt để Abstraction mang lại sự suy giảm đáng kể mức độ phụ thuộc chéo giữa các phân hệ, tạo không gian linh hoạt tối đa để thay thế hoặc luân chuyển nhà cung cấp. Sự phân tách này giúp hệ thống trở nên dễ kiểm thử, dễ bảo trì, và mở ra tiềm năng mở rộng không giới hạn trong tương lai. Hệ quả tất yếu là rủi ro kỹ thuật giảm thiểu tối đa, đặc biệt là khi các API bên ngoài thay đổi.

\vspace{0.15cm}
\noindent\textbf{Cân bằng mức độ trừu tượng:} Dù mang lại giá trị to lớn, nhóm phát triển ý thức được rằng trừu tượng hóa không phải là công cụ lạm dụng bừa bãi. Việc trừu tượng hóa quá mức sẽ biến hệ thống thành một cấu trúc mơ hồ, trong khi quá ít lại khiến logic nghiệp vụ bị gắn chặt với các nền tảng. Nhóm đã thận trọng lựa chọn điểm cân bằng: \textit{mọi lớp trừu tượng được tạo ra đều phải gắn liền với một định hướng kiến trúc rõ ràng}. Mỗi lớp trung gian chỉ tồn tại nếu nó thực sự giải quyết bài toán thay thế nhà cung cấp, chuẩn hóa dữ liệu, hoặc gia tăng khả năng mở rộng.

\subsection{Main Algorithmic Workflows}

Thiết kế thuật toán của hệ thống được tập trung vào việc giải quyết quy trình từ khi nhận yêu cầu thô đến lúc xuất ra kết quả tinh chỉnh. Dưới đây là các luồng làm việc (workflow) chính được thiết kế tối ưu và minh họa bằng lưu đồ phân lớp.

\vspace{0.3cm}
\noindent\textbf{1. AI Consultation Workflow (Điều phối AI linh hoạt)}
\begin{itemize}
    \item \textbf{Goal:} Chuyển đổi truy vấn ngôn ngữ tự nhiên của người dùng thành kết quả hành động và phản hồi thân thiện.
    \item \textbf{Input:} Câu chat của người dùng và ngữ cảnh hội thoại trước đó.
    \item \textbf{Main processing steps:} Giải thuật điều phối đi qua 4 lớp nhằm đảm bảo tính tất định cho logic nghiệp vụ.
    \item \textbf{Output:} Phản hồi dạng Markdown kèm theo dữ liệu bản đồ để frontend vẽ marker trực quan.
    \item \textbf{Edge cases:} Nếu không trích xuất được ý định hợp lệ, hệ thống fallback về luồng \texttt{GENERAL\_CHAT} mặc định để yêu cầu người dùng làm rõ.
\end{itemize}

\noindent\textbf{3. RAG Workflow (Truy xuất bổ sung ngữ cảnh)}
\begin{itemize}
    \item \textbf{Goal:} Rút trích các tri thức ẩn từ dữ liệu dạng văn bản (như bình luận, mô tả) để đưa cho LLM tham khảo.
    \item \textbf{Input:} Truy vấn cần phân tích sâu (ví dụ: "Có khách sạn nào yên tĩnh không?").
    \item \textbf{Main processing steps:} Sinh embedding từ truy vấn $\to$ Tìm kiếm Cosine similarity qua \texttt{pgvector} $\to$ Tiêm (Inject) chunk ngữ cảnh vào prompt.
    \item \textbf{Output:} Khối ngữ cảnh văn bản được nhúng chìm vào luồng hội thoại.
    \item \textbf{Edge cases:} Nếu điểm tương đồng tối đa thấp hơn ngưỡng (threshold), hệ thống kết luận không có dữ liệu nội bộ liên quan để tránh nhiễu prompt.
\end{itemize}

\subsection{Failure Handling and Fallback Design}

Một hệ thống phân tán mạnh cần có khả năng sinh tồn khi các yếu tố bên ngoài sụp đổ. Thiết kế xử lý lỗi (Failure Handling) của hệ thống xoay quanh nguyên tắc "Lỗi một phần không làm sập toàn bộ":

\begin{itemize}
    \item [\faNetworkWired] \textbf{Chuỗi dự phòng API (Fallback Chain):} Trong truy vấn bản đồ, nếu nguồn cung cấp chính bị lỗi (HTTP 500 hoặc Timeout), hệ thống lập tức nhảy sang bộ chuyển tiếp API dự phòng. 
    \item [\faGhost] \textbf{Xử lý Dữ liệu Rỗng (Graceful Empty States):} Nếu workflow tìm kiếm trả về danh sách rỗng, LLM được thiết lập bằng System Prompt cứng để "tuyệt đối không bịa đặt địa điểm (No Hallucination)". Nó sẽ thành thực báo cáo và đưa ra gợi ý mở rộng vùng tìm kiếm.
    \item [\faLock] \textbf{Cách ly Node Lỗi trong Workflow:} Nhờ kiến trúc chia nhỏ các công cụ (tools), nếu công cụ gợi ý sâu thất bại, hệ thống vẫn giữ nguyên được kết quả của công cụ tìm kiếm cơ bản và thông báo sự cố cục bộ đến frontend.
\end{itemize}

\subsection{Design Summary}

Thiết kế System và Algorithm của dự án chứng minh rằng việc kết hợp nhiều công nghệ tiên tiến (AI, Vector DB, Map Services) không đơn thuần là quá trình nối ghép API, mà phải là nghệ thuật tổ chức hệ thống. Nhờ tuân thủ mô hình C4 để phân rã chức năng, và phân tách các workflow phức tạp thành DAG các thuật toán đơn nhiệm, dự án đạt được:
\begin{itemize}
    \item [\faVial] \textbf{Tính kiểm thử (Testability):} Các logic xếp hạng và khử trùng lặp có thể Unit Test độc lập.
    \item [\faPlug] \textbf{Tính thay thế (Replaceability):} Khả năng dễ dàng cắm-rút các Provider AI hoặc Bản đồ.
    \item [\faHeartbeat] \textbf{Tính kiên cường (Resilience):} Xử lý lỗi tinh tế giúp hệ thống không rơi vào điểm chết khi đối mặt với sự cố mạng bên ngoài.
\end{itemize}

\subsection{Di trú Kiến trúc: Microservices $\to$ Modular Monolith}

Hệ thống nguyên bản vận hành với bốn dịch vụ riêng rẽ (Nginx Gateway, Core-service, AI-service, Recommendation-service). Kiến trúc này bộc lộ các vấn đề:

\begin{itemize}
  \item \textbf{Chi phí mạng:} Mỗi yêu cầu chat trải qua ba chặng HTTP, gây độ trễ tích lũy.
  \item \textbf{Trùng lặp logic:} Xác thực, xử lý lỗi, cấu hình bị sao chép giữa các dịch vụ.
  \item \textbf{Phân mảnh kiểu:} Python $\leftrightarrow$ TypeScript yêu cầu tuần tự hóa thủ công, dễ sai sót.
  \item \textbf{Phức tạp vận hành:} Bốn Docker image, bốn pipeline triển khai riêng biệt.
\end{itemize}

\textbf{Giải pháp:} Hợp nhất thành một monolith dạng module NestJS duy nhất. Các lời gọi HTTP liên dịch vụ được thay thế bằng phương thức gọi trực tiếp qua dependency injection, loại bỏ Gateway Nginx trung gian.

\subsection{Các Thách thức và Giải pháp}

\subsubsection{Tính phi tất định của LLM}

\textbf{Thách thức:} LLM sinh phản hồi không nhất quán về định dạng, dễ hallucination dữ liệu, và độ trễ khó dự đoán.

\textbf{Giải pháp --- Kiến trúc Bốn Lớp:}
\begin{itemize}
  \item \textbf{Lớp 1 (Router):} LLM với JSON Mode $\to$ phân loại ý định, trích xuất tham số có cấu trúc. Thất bại $\to$ mặc định GENERAL\_CHAT.
  \item \textbf{Lớp 2 (Workflow Engine):} Thực thi DAG các bước tất định, không gọi LLM.
  \item \textbf{Lớp 3 (Tool Layer):} Mỗi công cụ có hợp đồng đầu ra chuẩn hóa, không ném ngoại lệ.
  \item \textbf{Lớp 4 (Composer):} Neo phản hồi Markdown vào dữ liệu thực tế từ công cụ, giảm hallucination.
\end{itemize}

Lợi ích: Workflow mới có thể thêm dưới dạng JSON, không cần biên dịch lại.

\subsubsection{Streaming SSE qua POST}

\textbf{Thách thức:} SSE chuẩn chỉ hỗ trợ GET, trong khi endpoint chat cần gửi JSON payload và token xác thực.

\textbf{Giải pháp --- Mô hình hai pha:}
\begin{itemize}
  \item Sử dụng \texttt{fetch()} với \texttt{ReadableStream} thay vì \texttt{EventSource}, cho phép POST và đính kèm token.
  \item Pha 1: Gửi dữ liệu cấu trúc (\texttt{searchAction}) ngay sau khi công cụ thực thi --- cho phép frontend render bản đồ trước.
  \item Pha 2: Truyền token LLM dưới dạng delta SSE.
  \item Header \texttt{X-Accel-Buffering: no} để vô hiệu hóa bộ đệm Nginx khi triển khai sau reverse proxy.
\end{itemize}

\subsubsection{Tích hợp Đa Nhà cung cấp Bản đồ}

\textbf{Thách thức:} Mỗi API bản đồ trả về định dạng khác nhau; lỗi một nhà cung cấp không được ảnh hưởng toàn bộ tìm kiếm.

\textbf{Giải pháp:}
\begin{itemize}
  \item Chuẩn hóa giao diện đầu ra cho mọi nhà cung cấp.
  \item Gọi song song qua \texttt{Promise.all} với \texttt{.catch()} riêng biệt --- đảm bảo cách ly lỗi.
  \item Chuỗi dự phòng: Goong $\to$ OSM $\to$ mảng rỗng.
\end{itemize}

\subsubsection{Vector Embedding và Prisma}

\textbf{Thách thức:} Prisma chưa hỗ trợ nguyên sinh kiểu \texttt{vector} của PostgreSQL.

\textbf{Giải pháp:} Khai báo cột \texttt{embedding} dưới dạng \texttt{Unsupported("vector")}, thao tác qua SQL thô (\texttt{\$queryRawUnsafe}).

\subsubsection{Xác thực Firebase}

\textbf{Thách thức:} Tích hợp Firebase JWT vào vòng đời NestJS guard mà không sao chép logic xác thực trên mọi controller.

\textbf{Giải pháp:} Triển khai guard với \texttt{CanActivate}: trích xuất Authorization header $\to$ xác thực token qua Firebase Admin SDK $\to$ gắn decoded token vào request. Frontend Axios interceptor tự động đính kèm token.

\subsection{Các Thách thức Khác}

\textbf{Xử lý đồng thời:} Ba mẫu thiết kế được áp dụng: (1) \texttt{Promise.all} với catch riêng biệt; (2) ghi vào khóa khác nhau trong shared state để tránh race condition; (3) Active Flag pattern để tránh cập nhật state sau khi component unmount.

\textbf{Triển khai Docker:} Ba container (frontend/Nginx, backend/Node 22, PostgreSQL/pgvector) với chiến lược build đa tầng tận dụng layer cache. Docker override cho môi trường phát triển sử dụng volume ẩn danh tránh xung đột \texttt{node\_modules}.

\textbf{Quản lý bản đồ frontend:} Chuyển đổi layer bằng opacity, MarkerClusterGroup với chunkedLoading cho hiệu năng, CustomEvent thay vì prop drilling cho tích hợp AI-Search.

\subsection{Kết luận}

Quá trình triển khai cho thấy việc chuyển đổi từ mô hình thiết kế sang mã nguồn đòi hỏi liên tục giải quyết các vấn đề thực tế: tính phi tất định của LLM, hạn chế của ORM với kiểu dữ liệu đặc thù, và đảm bảo tính nhất quán xuyên suốt các tầng. Mười hai vấn đề đã được ghi nhận với các mức độ ưu tiên khác nhau.

\subsection{Cải tiến Sau Kiểm thử}

Quá trình kiểm thử đã phát hiện và dẫn đến các cải tiến sau:

\begin{itemize}
  \item \textbf{Cải tiến xử lý lỗi:} Chuẩn hóa phản hồi lỗi thành cấu trúc JSON thống nhất với timestamp, loại bỏ try/catch trùng lặp ở controller.
  \item \textbf{Cải tiến khử trùng lặp:} Bổ sung khóa mờ theo tọa độ (lat:lng độ chính xác 3 chữ số thập phân) bên cạnh khóa chính xác, giúp phát hiện trùng lặp khi ID nhà cung cấp khác nhau.
  \item \textbf{Cải tiến buffer SSE:} Cả backend và frontend được đồng bộ buffer pattern để xử lý chunk SSE bị phân mảnh do TCP.
  \item \textbf{Cải tiến Presence:} Thêm kiểm tra tải 100 người dùng đồng thời, đảm bảo cấu trúc dữ liệu hoạt động ổn định dưới tải cao.
  \item \textbf{Cải tiến NLP:} Tham số hóa 14 văn bản tiếng Việt thực tế, bao phủ đa dạng biến thể ngôn ngữ (tên tỉnh thành, loại hình, mức giá).
\end{itemize}

\end{itemize}
\subsection{Những kiến thức và kỹ năng đạt được}
Môn học và đồ án lần này đã mang lại cho chúng em những bài học vô cùng thực tế, giúp thu hẹp khoảng cách giữa lý thuyết và ứng dụng:
\begin{itemize}
  \item \textbf{Vận dụng 4 trụ cột cốt lõi của Tư duy Máy tính:} Thông qua đồ án, chúng em đã làm chủ và áp dụng thành công mô hình tư duy này để giải quyết bài toán thực tế. Cụ thể, chúng em thực hiện 
  \begin{enumerate}
    \item \textit{Phân rã bài toán} (Decomposition) để chia nhỏ hệ thống thành các module độc lập (\textit{AiModule} và \textit{RecommendationsModule});
    \item \textit{Nhận biết mẫu} (Pattern Recognition) để tìm ra khuôn mẫu trong hành vi tìm kiếm của khách hàng;
    \item \textit{Trừu tượng hóa} (Abstraction) để lọc bỏ thông tin nhiễu, trích xuất tham số cốt lõi qua \textit{Intent Parsing};
    \item \textit{Thiết kế thuật toán} (Algorithm Design) để xây dựng luồng vận hành tối ưu cho Chatbot.
  \end{enumerate}
    \noindent Các kỹ năng này giúp chúng em định hình được tư duy logic vững chắc cho các dự án phức tạp sau này.
  
  \item \textbf{Tư duy khai thác Trí tuệ nhân tạo: }Chúng em đã học được cách viết kỹ năng Prompt Engineering và cách ứng dụng AI một cách hiệu quả, biến AI thành một công cụ đồng hành đắc lực để tối ưu hóa việc tìm kiếm thông tin thay vì lạm dụng một cách máy móc.
  \item \textbf{Làm quen với các công nghệ mới: }Đồ án là cơ hội để chúng em tiếp cận và biết cách sử dụng hàng loạt công cụ, ngôn ngữ và nền tảng cốt lõi trong phát triển phần mềm hiện đại như HTML, JavaScript, nền tảng cơ sở dữ liệu Firebase, bản đồ OpenStreetMap,... Dù thời gian có hạn và chưa thể tìm hiểu quá sâu sắc từng công nghệ, nhưng đây là nền móng cực kỳ quan trọng cho các môn học chuyên ngành tiếp theo.
  \item \textbf{Kỹ năng làm việc nhóm và Quản lý mã nguồn: }Biết cách sử dụng GitHub để quản lý các phiên bản code, phối hợp phân chia công việc trong nhóm một cách hệ thống.
  \item \textbf{Tư duy Hệ thống: }Hiểu rõ bức tranh toàn cảnh về cách một hệ thống phần mềm vận hành trong thực tế (cách Frontend tương tác với Backend, cách gọi API và truy vấn dữ liệu từ Database).
\end{itemize}

\subsection{Hướng phát triển trong tương lai}
Nhằm nâng cấp và hoàn thiện dự án thành một sản phẩm có tính ứng dụng thực tế cao hơn, các mục tiêu trong tương lai bao gồm:
\begin{itemize}
  \item \textbf{Nghiên cứu sâu về RAG và LLM: }Tối ưu hóa sâu hơn kỹ thuật Retrieval-Augmented Generation (RAG) kết hợp với các mô hình ngôn ngữ lớn (như Groq) để nâng cao độ chính xác, giảm thiểu hiện tượng "ảo tưởng" (hallucination) của Chatbot khi trả lời khách hàng.
  \item \textbf{Tối ưu hóa UI/UX: }Thiết kế lại giao diện người dùng thân thiện, trực quan hơn trên cả hai nền tảng Web và Mobile, tối ưu hóa các bộ lọc nâng cao để nâng cao trải nghiệm của các nhóm đối tượng Target Users.
\end{itemize}
